194 void learn_metric(std::mt19937_64& rng) {
195 const int i = warmup_counter_++;
196 if (i >= windows_.init_buffer && i < warmup_total_ - windows_.term_buffer) estimator_.add(q_);
197 if (!windows_.closes(i + 1)) return;
198 const std::vector<double> est = estimator_.regularised();
199 std::vector<double> cov(n_ * n_, 0.0);
200 if (estimator_.dense()) {
201 cov = est;
202 } else {
203 for (std::size_t a = 0; a < n_; ++a) cov[a * n_ + a] = est[a];
204 }
205 for (double v : cov)
206 if (!std::isfinite(v))
207 throw SamplerConfigurationError("nuts: numerical overflow in metric adaptation (extreme values on the unconstrained space)");
208 set_metric(cov);
209 init_step_size(rng);
210 if (adapting_) adapt_.restart(eps_); // mu = log(10 * the new step size), counters restarted
211 estimator_.restart();
212 }

214 void set_metric(const std::vector<double>& cov) {
215 S_.assign(n_ * n_, 0.0);
216 L_.assign(n_ * n_, 0.0);
217 if (cov.empty()) {
218 for (std::size_t i = 0; i < n_; ++i) { S_[i * n_ + i] = 1.0; L_[i * n_ + i] = 1.0; }
219 return;
220 }
221 if (cov.size() != n_ * n_) throw SamplerConfigurationError("nuts: inverse_metric must be dim x dim");
222 S_ = cov;
223 for (std::size_t i = 0; i < n_; ++i)
224 for (std::size_t j = 0; j <= i; ++j) {
225 double s = S_[i * n_ + j];
226 for (std::size_t k = 0; k < j; ++k) s -= L_[i * n_ + k] * L_[j * n_ + k];
227 if (i == j) {
228 if (!(s > 0.0)) throw SamplerConfigurationError("nuts: inverse_metric is not positive definite");
229 L_[i * n_ + i] = std::sqrt(s);
230 } else {
231 L_[i * n_ + j] = s / L_[j * n_ + j];
232 }
233 }

237 void init_step_size(std::mt19937_64& rng) {
238 const std::vector<double> q0 = q_, g0 = grad_;
239 const double lp0 = lp_;
240 std::vector<double> p(n_);
241 auto one_step_delta = [&]() {
242 q_ = q0; grad_ = g0; lp_ = lp0;
243 sample_momentum(p, rng);
244 const double H0 = hamiltonian(lp_, p);
245 leapfrog(q_, p, grad_, lp_, eps_);
246 double h = hamiltonian(lp_, p);
247 if (std::isnan(h)) h = std::numeric_limits<double>::infinity();
248 return H0 - h;
249 };
250 double delta_H = one_step_delta();
251 const int direction = delta_H > std::log(0.8) ? 1 : -1;
252 for (;;) {
253 delta_H = one_step_delta();
254 if (direction == 1 && !(delta_H > std::log(0.8))) break;
255 if (direction == -1 && !(delta_H < std::log(0.8))) break;
256 eps_ = direction == 1 ? 2.0 * eps_ : 0.5 * eps_;
257 if (eps_ > 1e7) throw SamplerConfigurationError("nuts: the posterior is improper (step size diverged)");
258 if (eps_ == 0.0) throw SamplerConfigurationError("nuts: no acceptably small step size");
259 }
260 q_ = q0; grad_ = g0; lp_ = lp0;
261 }

263 static double log_sum_exp(double a, double b) {
264 if (a == -std::numeric_limits<double>::infinity()) return b;
265 if (b == -std::numeric_limits<double>::infinity()) return a;
266 const double m = std::max(a, b);
267 return m + std::log(std::exp(a - m) + std::exp(b - m));
268 }
269 static bool criterion(const std::vector<double>& ps_minus, const std::vector<double>& ps_plus, const std::vector<double>& rho) {
270 double a = 0.0, b = 0.0;
271 for (std::size_t i = 0; i < rho.size(); ++i) { a += ps_plus[i] * rho[i]; b += ps_minus[i] * rho[i]; }
272 return a > 0.0 && b > 0.0;
273 }
274 std::vector<double> dtau_dp(const std::vector<double>& p) const {
275 std::vector<double> v(n_, 0.0);
276 for (std::size_t i = 0; i < n_; ++i) { double s = 0.0; for (std::size_t j = 0; j < n_; ++j) s += S_[i * n_ + j] * p[j]; v[i] = s; }
277 return v;
278 }
279 double hamiltonian(double lp, const std::vector<double>& p) const {
280 const std::vector<double> v = dtau_dp(p);
281 double k = 0.0;
282 for (std::size_t i = 0; i < n_; ++i) k += p[i] * v[i];
283 return -lp + 0.5 * k;
284 }
286 void sample_momentum(std::vector<double>& p, std::mt19937_64& rng) const {
287 std::normal_distribution<double> N(0.0, 1.0);
288 std::vector<double> u(n_);
289 for (double& v : u) v = N(rng);
290 for (std::size_t ii = n_; ii-- > 0;) {
291 double s = u[ii];
292 for (std::size_t k = ii + 1; k < n_; ++k) s -= L_[k * n_ + ii] * p[k];
293 p[ii] = s / L_[ii * n_ + ii];
294 }
295 }
296 void leapfrog(std::vector<double>& q, std::vector<double>& p, std::vector<double>& grad, double& lp, double eps) {
297 for (std::size_t i = 0; i < n_; ++i) p[i] += 0.5 * eps * grad[i]; // grad of log density = -dphi/dq
298 const std::vector<double> v = dtau_dp(p);
299 for (std::size_t i = 0; i < n_; ++i) q[i] += eps * v[i];
300 lp = f_(q, grad);
301 ++evaluations_;
302 if (!std::isfinite(lp)) { lp = -std::numeric_limits<double>::infinity(); return; }
303 for (std::size_t i = 0; i < n_; ++i) p[i] += 0.5 * eps * grad[i];
304 }
305 bool build_tree(int depth, State& z, State& z_propose, std::vector<double>& ps_beg, std::vector<double>& ps_end,
306 std::vector<double>& rho, std::vector<double>& p_beg, std::vector<double>& p_end, double H0, double sign,
307 int& n_leapfrog, double& log_sum_weight, double& sum_metro_prob, std::mt19937_64& rng) {
308 if (depth == 0) {
309 leapfrog(z.q, z.p, z.grad, z.lp, sign * eps_);
310 ++n_leapfrog;
311 double h = std::isfinite(z.lp) ? hamiltonian(z.lp, z.p) : std::numeric_limits<double>::infinity();
312 if (std::isnan(h)) h = std::numeric_limits<double>::infinity();
313 if (h - H0 > opt_.max_delta_h) divergent_ = true;
314 log_sum_weight = log_sum_exp(log_sum_weight, H0 - h);
315 sum_metro_prob += H0 - h > 0.0 ? 1.0 : std::exp(H0 - h);
316 z_propose = z;
317 ps_beg = dtau_dp(z.p);
318 ps_end = ps_beg;
319 for (std::size_t i = 0; i < n_; ++i) rho[i] += z.p[i];
320 p_beg = z.p;
321 p_end = p_beg;
322 return !divergent_;
323 }
324 double log_sum_weight_init = -std::numeric_limits<double>::infinity();
325 std::vector<double> p_init_end(n_, 0.0), ps_init_end(n_, 0.0), rho_init(n_, 0.0);
326 if (!build_tree(depth - 1, z, z_propose, ps_beg, ps_init_end, rho_init, p_beg, p_init_end, H0, sign, n_leapfrog,
327 log_sum_weight_init, sum_metro_prob, rng))
328 return false;
329 State z_propose_final = z;
330 double log_sum_weight_final = -std::numeric_limits<double>::infinity();
331 std::vector<double> p_final_beg(n_, 0.0), ps_final_beg(n_, 0.0), rho_final(n_, 0.0);
332 if (!build_tree(depth - 1, z, z_propose_final, ps_final_beg, ps_end, rho_final, p_final_beg, p_end, H0, sign, n_leapfrog,
333 log_sum_weight_final, sum_metro_prob, rng))
334 return false;
335 const double log_sum_weight_subtree = log_sum_exp(log_sum_weight_init, log_sum_weight_final);
336 log_sum_weight = log_sum_exp(log_sum_weight, log_sum_weight_subtree);
337 std::uniform_real_distribution<double> U(0.0, 1.0);
338 if (log_sum_weight_final > log_sum_weight_subtree) {
339 z_propose = z_propose_final;
340 } else {
341 const double accept_prob = std::exp(log_sum_weight_final - log_sum_weight_subtree);
342 if (U(rng) < accept_prob) z_propose = z_propose_final;
343 }
344 std::vector<double> rho_subtree(n_);
345 for (std::size_t i = 0; i < n_; ++i) { rho_subtree[i] = rho_init[i] + rho_final[i]; rho[i] += rho_subtree[i]; }
346 bool persist = criterion(ps_beg, ps_end, rho_subtree);
347 std::vector<double> rho_ext(n_);
348 for (std::size_t i = 0; i < n_; ++i) rho_ext[i] = rho_init[i] + p_final_beg[i];
349 persist = persist && criterion(ps_beg, ps_final_beg, rho_ext);
350 for (std::size_t i = 0; i < n_; ++i) rho_ext[i] = rho_final[i] + p_init_end[i];
351 persist = persist && criterion(ps_init_end, ps_end, rho_ext);
352 return persist;
353 }
