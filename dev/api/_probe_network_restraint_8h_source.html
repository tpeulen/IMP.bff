63 /* `avs_` holds bare AV decorator handles, which are views onto particles
64 that the Model owns, not state of their own. They are therefore stored
65 as the particle indices they decorate and rebuilt against the restored
66 model on load, rather than archived as pointers. */
67 template<class Archive> void serialize(Archive &ar) {
68 ar(cereal::base_class<IMP::Restraint>(this),
69 n_samples, av_pi_, model_ps_, distances_,
70 space_fixed_, shared_map_, distance_, quad_k_, search_grid_factor_,
71 search_stencil_, search_mode_, points_, atom_points_);
72 // On save this is built from avs_; on load ar() overwrites it and the
73 // decorators are rebuilt from it below. A single serialize() (rather
74 // than a save/load pair) is required here because IMP::Restraint
75 // already supplies one, and cereal rejects two candidate functions.
76 std::map<std::string, IMP::ParticleIndex> av_index;
77 for (const auto &kv : avs_) {
78 av_index[kv.first] = kv.second->get_particle_index();
79 }
80 ar(av_index);
81 if (std::is_base_of<cereal::detail::InputArchiveBase, Archive>::value) {
82 avs_.clear();
83 for (const auto &kv : av_index) {
84 avs_[kv.first].reset(new IMP::bff::ProbeAccessibleVolumeDecorator(get_model(), kv.second));
85 }
86 registry_ = nullptr;
87 configure_avs();
88 }
89 }
