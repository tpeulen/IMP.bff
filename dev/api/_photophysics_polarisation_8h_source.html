110template <typename T = double>
111inline void isotropic_from_polarised(const T* vv, const T* vh, std::size_t n, const T& g,
112 const T& w_vv, const T& w_vh, T* out) {
113 const T den = w_vv - w_vh;
114 for (std::size_t i = 0; i < n; ++i) out[i] = (w_vv * g * vh[i] - w_vh * vv[i]) / den;
115}

156template <typename T = double>
157inline void anisotropy_from_polarised(const T* vv, const T* vh, std::size_t n,
158 const T& g, const T& w_vv, const T& w_vh,
159 T* out, double floor = 1e-300) {
160 for (std::size_t i = 0; i < n; ++i) {
161 // (w_vv - w_vh) divides out of numerator and denominator alike, so this
162 // is r = R / S written without forming either
163 const T scaled_iso = w_vv * g * vh[i] - w_vh * vv[i];
164 out[i] = (scaled_iso > T(floor)) ? (vv[i] - g * vh[i]) / scaled_iso : T(0.0);
165 }
166}
