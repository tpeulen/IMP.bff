362inline void bayesian_softmax_inplace(std::vector<double>& x) {
363 double mx = -1e300;
364 for (double t : x) mx = std::max(mx, t);
365 double s = 0.0;
366 for (double& t : x) { t = std::exp(t - mx); s += t; }
367 for (double& t : x) t /= s;

370inline BayesianDecayValues bayesian_decay_unpack(const BayesianDecayExperiment& f, const std::vector<double>& theta) {
371 //: through BayesianTransforms.h; the sum-to-zero basis is the experiment's own
372 //: (`Q_<name>`), since theta is defined on it
373 BayesianDecayValues out;
374 for (const BayesianDecayVariable& var : f.variables) {
375 const std::string& name = var.name, & tr = var.transform;
376 const double* z = theta.data() + var.offset;
377 std::vector<double> x;
378 if (tr == "identity") { x.resize(var.size); BayesianIdentityTransform<double>().to_constrained(z, var.size, x.data()); }
379 else if (tr == "log") { x.resize(var.size); BayesianLogTransform<double>().to_constrained(z, var.size, x.data()); }
380 else if (tr == "logit") { x.resize(var.size); BayesianLogitTransform<double>(var.lo, var.hi).to_constrained(z, var.size, x.data()); }
381 else if (tr == "alr") { x.resize(var.size + 1); BayesianALRTransform<double>(var.size + 1).to_constrained(z, var.size, x.data()); }
382 else if (tr == "sum_to_zero") {
383 BayesianSumToZeroTransform<double> t;
384 t.n = var.size + 1;
385 t.Q = f["Q_" + name].d;
386 x.resize(t.n);
387 t.to_constrained(z, var.size, x.data());
388 } else { throw std::runtime_error("bayesian_decay_unpack: transform " + tr + " of " + name); }
389 out[name] = std::move(x);
390 }
391 for (const auto& kv : f.fixed_values)
392 if (!out.count(kv.first)) out[kv.first] = kv.second;
393 return out;
394}

498 const double* M0 = e.S_Ag->d.data() + (j * nA + l) * K * Kint;
499 for (std::size_t k = 0; k < K; ++k) {
500 double t = 0.0; const double* row = M0 + k * Kint;
501 for (std::size_t i = 0; i < Kint; ++i) t += row[i] * spec[i];
502 qa[k] += ujl * t;
503 }
504 for (std::size_t r = 0; r < e.nrho_a; ++r) {
505 const double wr = w_rho_a[r] * ujl;
506 const double* Mr = e.S_Ag_rot_r->d.data() + ((r * nR + j) * nA + l) * K * Kint;
507 for (std::size_t k = 0; k < K; ++k) {
508 double t = 0.0; const double* row = Mr + k * Kint;
509 for (std::size_t i = 0; i < Kint; ++i) t += row[i] * spec[i];
510 qar[k] += wr * t;
511 }
512 }
513 }
514 }
515 double direct = 0.0;
516 if (!sc.direct.empty()) {
517 const std::string d = sc.direct;
518 direct = d == "EX_AG*n_da*per_mol" ? EX_AG * n_da * per_mol : d == "n_da*per_mol" ? n_da * per_mol
519 : d == "EX_AG" ? EX_AG : d == "1" ? 1.0 : 0.0;
520 }
521 for (std::size_t k = 0; k < K; ++k) {
522 S.acc[k] = QY_A * (qa[k] + a_dir[k] * direct);
523 S.accr[k] = QY_A * (qar[k] + a_dir_r[k] * direct);
524 }
525 }
526 sp[si_] = std::move(S);
527 });

530 const std::vector<std::string>& keys = f.keys;

532 std::map<std::string, const BayesianDecayPart*> part_of;
533 for (auto& pt : f.parts) part_of[pt.channel] = &pt;
534 for (std::size_t ic = 0; ic < keys.size(); ++ic) {

536 const BayesianDecayScope& sc = f.scopes[pt.scope];
537 const Species& S = sp[pt.scope];
538 const bool vh = pt.pol == 1, red = pt.colour == 1;
539 //: the polarised channel weight through PhotophysicsPolarisation.h: 2 - 3 l1 (VV), -1 + 3 l2 (VH)
540 const double sign = anisotropy_weight(vh ? POL_VH : POL_VV, bayesian_decay_scalar(v, "l1"), bayesian_decay_scalar(v, "l2"));
541 const double gf = vh ? (red && v.count("g_r") ? bayesian_decay_scalar(v, "g_r") : bayesian_decay_scalar(v, "g")) : 1.0;
542 const bool is_ref = sc.r0 == "r0_ref";
543 const std::vector<double>& Srho = is_ref ? Srho_ref : Srho_d;
544 const double r0 = bayesian_decay_scalar(v, sc.r0), r0_a = bayesian_decay_scalar(v, "r0_a");
545 std::vector<double> mix(K);
546 const double Gc = red ? bayesian_decay_scalar(v, "G_RED") : bayesian_decay_scalar(v, "G_GREEN");
547 const double CD = red ? bayesian_decay_scalar(v, "C_RD") : bayesian_decay_scalar(v, "C_GD"), CA = red ? bayesian_decay_scalar(v, "C_RA") : bayesian_decay_scalar(v, "C_GA");
548 //: donor and acceptor each mixed by PhotophysicsPolarisation.h's mix_polarised
549 //: ((iso + w r0 aniso) / g_channel, g applied once below)
550 std::vector<double> rot(K, 0.0), pol_d(K), pol_a(K);
551 for (std::size_t k = 0; k < K; ++k)
552 for (std::size_t i = 0; i < Kint; ++i) rot[k] += Srho[k * Kint + i] * S.don[1 + i];
553 mix_polarised(S.don.data(), rot.data(), K, r0, sign, 1.0, pol_d.data());
554 mix_polarised(S.acc.data(), S.accr.data(), K, r0_a, sign, 1.0, pol_a.data());
555 //: the emission: the detector's column of the crosstalk matrix, (donor, acceptor) =
556 //: (G C_D, G C_A), applied through internal/CrosstalkMixing.h, then g once
557 const double column[2] = {Gc * CD, Gc * CA};
558 std::vector<double> sources(2 * K);
559 std::copy(pol_d.begin(), pol_d.end(), sources.begin());
560 std::copy(pol_a.begin(), pol_a.end(), sources.begin() + std::ptrdiff_t(K));
561 internal::crosstalk_mix(column, 2, 1, sources.data(), K, mix.data());
562 for (std::size_t k = 0; k < K; ++k) mix[k] /= gf;
563 // the instrument stage (internal/TCSPCInstrument.h) in the basis's amplitude space:
564 // every column sums to one, so the response and the flat background are unit
565 // vectors -- the same stage TCSPCDecay applies in channel space
566 const std::string samp = pt.sample, chan = pt.channel;
567 const internal::TCSPCInstrumentSettings ins = bayesian_decay_instrument_settings(K);
568 internal::TCSPCInstrumentParameters<double> ip;
569 ip.scale = std::exp(bayesian_decay_scalar(v, "log_scale_" + samp));
570 ip.scatter = bayesian_decay_scalar(v, "scat_" + chan);
571 ip.background = bayesian_decay_scalar(v, "bkg_" + chan);
572 internal::tcspc_instrument_components(mix.data(), K, ins, ip, mix.data());
573 for (std::size_t k = 0; k < K; ++k) out[ic * K + k] = mix[k];
574 }
575 return out;

579// the analytic amplitude Jacobian: d a2 / d theta, (n_keys * K, dim)
580//
581// The port of s88.amplitude_jacobian, table-driven: the donor block is
582// (vv base + sum_j u_j S_R[j]) @ spectrum times QY_D and the scope's donor
583// excitation; the acceptor block is the sensitised sums plus a DIRECT term
584// whose partials in the spectrum, u and EX_AG follow from which of
585// EX_AG * n_da * per_mol it contains; then the mixing and the instrument,
586// both linear. A coordinate held by the graph has no column.

588struct BayesianDecayVariableInfo { std::size_t off = 0, size = 0; std::string transform; double lo = 0, hi = 1; bool present = false; };

590inline BayesianDecayVariableInfo bayesian_decay_variable_info(const BayesianDecayExperiment& f, const std::string& name) {
591 BayesianDecayVariableInfo vi;
592 const BayesianDecayVariable* var = f.variable(name);
593 if (var) { vi.off = var->offset; vi.size = var->size; vi.transform = var->transform; vi.lo = var->lo; vi.hi = var->hi; vi.present = true; }
594 return vi;
595}
596
598inline double bayesian_decay_dxdz(const BayesianDecayVariableInfo& vi, double x) {
599 if (vi.transform == "log") return x;
600 if (vi.transform == "logit") { const double sg = (x - vi.lo) / (vi.hi - vi.lo); return (vi.hi - vi.lo) * sg * (1.0 - sg); }
601 return 1.0;
602}

604inline std::vector<double> bayesian_decay_amplitude_jacobian(const BayesianDecayExperiment& f, const BayesianDecayTensors& e, const BayesianDecayValues& v, bool drop_softmax_norm = false) {
605 const std::size_t K = e.K, Kint = e.Kint, nR = e.nR, nA = e.nA, nra = e.nrho_a;
606 const std::size_t dim = f.dim, nkeys = f.keys.size();
607 const double ref_sd = f.ref_spec_sd;
608 auto VI = [&](const std::string& n) { return bayesian_decay_variable_info(f, n); };
609 const std::vector<double> p = bayesian_decay_distribution(e, v), spec = bayesian_decay_donor_spectrum(v);
610 const std::vector<double>& w_a = v.at("w_a"), &w_rho_a = v.at("w_rho_a");
611 const double x = bayesian_decay_scalar(v, "x_d0"), QY_D = bayesian_decay_scalar(v, "QY_D"), QY_A = bayesian_decay_scalar(v, "QY_A");
612 const double EX_AG = bayesian_decay_scalar(v, "EX_AG"), EX_DR = bayesian_decay_scalar(v, "EX_DR", 0.0);
613 double per_mol = 0.0; for (double t : spec) per_mol += t;
614 auto ex_of = [&](const std::string& s_) { return s_ == "1" ? 1.0 : s_ == "EX_DR" ? EX_DR : s_ == "EX_AG" ? EX_AG : 0.0; };
615 auto M = [&](std::size_t r, std::size_t c) { return r * dim + c; };

617 // spectrum derivatives: (Kint, dim)
618 std::vector<double> dspec(Kint * dim, 0.0), dspecr(Kint * dim, 0.0);
619 const BayesianDecayVariableInfo v_eps = VI("spec_eps");
620 if (v_eps.present) for (std::size_t i = 0; i < Kint; ++i) for (std::size_t j = 0; j < Kint; ++j)
621 dspec[i * dim + v_eps.off + j] = (i == j ? spec[i] : 0.0) - (drop_softmax_norm ? 0.0 : spec[i] * spec[j]);
622 std::vector<double> spec_ref;
623 const BayesianDecayVariableInfo v_tau = VI("log10_tau_ref");
624 if (v.count("log10_tau_ref") || v.count("spec_ref_eps")) {
625 spec_ref = bayesian_decay_reference_spectrum(e, v, ref_sd);
626 if (v_tau.present) {
627 const double mu = bayesian_decay_scalar(v, "log10_tau_ref");
628 std::vector<double> d(Kint); double sd_ = 0.0;
629 for (std::size_t i = 0; i < Kint; ++i) { d[i] = (std::log10(e.tau_c->d[i]) - mu) / (ref_sd * ref_sd); sd_ += spec_ref[i] * d[i]; }
630 for (std::size_t i = 0; i < Kint; ++i) dspecr[i * dim + v_tau.off] = spec_ref[i] * (d[i] - sd_) * bayesian_decay_dxdz(v_tau, mu);
631 }
632 }
633 // p, dp/dc through the spline and the sum-to-zero basis: (nR, dim)
634 const BayesianDecayVariableInfo v_c = VI("c");
635 const BayesianDecayArray& Q = f["Q_c"];
636 const std::size_t nc = e.spl->shape[1], nz = Q.shape[1];
637 std::vector<double> dp(nR * dim, 0.0);
638 {
639 std::vector<double> BQ(nR * nz, 0.0); // spl @ Q
640 for (std::size_t j = 0; j < nR; ++j) for (std::size_t a = 0; a < nc; ++a) {
641 const double s_ = e.spl->d[j * nc + a];
642 if (s_ == 0.0) continue;
643 for (std::size_t b = 0; b < nz; ++b) BQ[j * nz + b] += s_ * Q.d[a * nz + b];
644 }
645 std::vector<double> pBQ(nz, 0.0); // p^T spl Q
646 for (std::size_t j = 0; j < nR; ++j) for (std::size_t b = 0; b < nz; ++b) pBQ[b] += p[j] * BQ[j * nz + b];
647 for (std::size_t j = 0; j < nR; ++j) for (std::size_t b = 0; b < nz; ++b)
648 dp[j * dim + v_c.off + b] = p[j] * BQ[j * nz + b] - p[j] * pBQ[b];
649 }
650 const BayesianDecayVariableInfo v_x = VI("x_d0");
651 const double dx = v_x.present ? bayesian_decay_dxdz(v_x, x) : 0.0;
652 // ALR Jacobians: (n, n-1)
653 auto alr_jac = [&](const std::string& n) {
654 const std::vector<double>& w = v.at(n); const std::size_t nn = w.size();
655 std::vector<double> J(nn * (nn - 1));
656 for (std::size_t i = 0; i < nn; ++i) for (std::size_t j = 0; j + 1 < nn; ++j) J[i * (nn - 1) + j] = (i == j ? w[i] : 0.0) - w[i] * w[j];
657 return J;
658 };
659 const BayesianDecayVariableInfo v_wa = VI("w_a"), v_wra = VI("w_rho_a");
660 const std::vector<double> Jwa = alr_jac("w_a"), Jwra = alr_jac("w_rho_a");
661 // contracted tensors
662 std::vector<double> MR(nR * K, 0.0), MA(nR * nA * K, 0.0), MAr(nra * nR * nA * K, 0.0), base_spec(K, 0.0);
663 for (std::size_t k = 0; k < K; ++k) for (std::size_t i = 0; i < Kint; ++i) base_spec[k] += e.base->d[k * Kint + i] * spec[i];
664 for (std::size_t j = 0; j < nR; ++j) for (std::size_t k = 0; k < K; ++k) {
665 double t = 0.0; for (std::size_t i = 0; i < Kint; ++i) t += e.S_R->d[(j * K + k) * Kint + i] * spec[i]; MR[j * K + k] = t;
666 }
667 internal::bayesian_parallel_for(nR, [&](std::size_t j) { // in parallel over distances
668 for (std::size_t l = 0; l < nA; ++l) for (std::size_t k = 0; k < K; ++k) {
669 double t = 0.0; const double* row = e.S_Ag->d.data() + ((j * nA + l) * K + k) * Kint;
670 for (std::size_t i = 0; i < Kint; ++i) t += row[i] * spec[i];
671 MA[(j * nA + l) * K + k] = t;
672 for (std::size_t r = 0; r < nra; ++r) {
673 double tr = 0.0; const double* rr = e.S_Ag_rot_r->d.data() + (((r * nR + j) * nA + l) * K + k) * Kint;
674 for (std::size_t i = 0; i < Kint; ++i) tr += rr[i] * spec[i];
675 MAr[((r * nR + j) * nA + l) * K + k] = tr;
676 }
677 }
678 });
679 std::vector<double> a_dir(K, 0.0), a_dir_r(K, 0.0), A_dir_rot(nA * K, 0.0), dadirr_dwra(K * nra, 0.0);
680 for (std::size_t l = 0; l < nA; ++l) for (std::size_t k = 0; k < K; ++k) {
681 a_dir[k] += w_a[l] * e.A_dir->d[l * K + k];
682 for (std::size_t r = 0; r < nra; ++r) {
683 const double t = e.A_dir_rot_r->d[(r * nA + l) * K + k];
684 A_dir_rot[l * K + k] += w_rho_a[r] * t;
685 dadirr_dwra[k * nra + r] += w_a[l] * t;
686 }
687 }
688 for (std::size_t l = 0; l < nA; ++l) for (std::size_t k = 0; k < K; ++k) a_dir_r[k] += w_a[l] * A_dir_rot[l * K + k];

690 struct Blocks { std::vector<double> Dd, Da, Dr, don, acc, accr; };
691 const std::size_t n_scopes = f.scopes.size();
692 std::vector<Blocks> blk(n_scopes);
693 internal::bayesian_parallel_for(n_scopes, [&](std::size_t si_) {
694 const BayesianDecayScope& sc = f.scopes[si_];
695 Blocks B{std::vector<double>(K * dim, 0.0), std::vector<double>(K * dim, 0.0), std::vector<double>(K * dim, 0.0),
696 std::vector<double>(K, 0.0), std::vector<double>(K, 0.0), std::vector<double>(K, 0.0)};
697 const bool fret = sc.fret, sens = sc.sensitised;
698 std::vector<double> u(nR, 0.0), du_dx(nR, 0.0); double n_da = 0.0;
699 if (fret) for (std::size_t j = 0; j < nR; ++j) { u[j] = (1.0 - x) * p[j]; n_da += u[j]; du_dx[j] = -p[j] * dx; }
700 // --- donor ---
701 if (!sc.spectrum.empty()) {
702 const bool is_ref = sc.spectrum == "reference";
703 const std::vector<double>& S_use = is_ref ? spec_ref : spec;
704 const std::vector<double>& dS = is_ref ? dspecr : dspec;
705 const std::string exs = sc.don_ex; const double exd = ex_of(exs);
706 // Mdon = vv base + sum_j u_j S_R[j] (K, Kint)
707 std::vector<double> Mdon(K * Kint);
708 const double vv = fret ? x : 1.0;
709 for (std::size_t k = 0; k < K * Kint; ++k) Mdon[k] = vv * e.base->d[k];
710 if (fret) for (std::size_t j = 0; j < nR; ++j) { if (u[j] == 0.0) continue;
711 for (std::size_t k = 0; k < K * Kint; ++k) Mdon[k] += u[j] * e.S_R->d[j * K * Kint + k]; }
712 std::vector<double> q(K, 0.0);
713 for (std::size_t k = 0; k < K; ++k) for (std::size_t i = 0; i < Kint; ++i) q[k] += Mdon[k * Kint + i] * S_use[i];
714 for (std::size_t k = 0; k < K; ++k) B.don[k] = QY_D * exd * q[k];
715 for (std::size_t k = 0; k < K; ++k) for (std::size_t i = 0; i < Kint; ++i) {
716 const double mk = QY_D * exd * Mdon[k * Kint + i];
717 if (mk == 0.0) continue;
718 for (std::size_t c = 0; c < dim; ++c) B.Dd[M(k, c)] += mk * dS[i * dim + c];
719 }
720 const std::vector<double> qspec_base = [&] { std::vector<double> t(K, 0.0);
721 for (std::size_t k = 0; k < K; ++k) for (std::size_t i = 0; i < Kint; ++i) t[k] += e.base->d[k * Kint + i] * S_use[i]; return t; }();
722 if (fret) {
723 // through u (c and x) and vv = x; MR contracted with the donor spectrum (fret scopes carry it)
724 for (std::size_t k = 0; k < K; ++k) {
725 for (std::size_t j = 0; j < nR; ++j) {
726 const double mr = QY_D * exd * MR[j * K + k];
727 for (std::size_t b = 0; b < nz; ++b) B.Dd[M(k, v_c.off + b)] += mr * (1.0 - x) * dp[j * dim + v_c.off + b];
728 if (v_x.present) B.Dd[M(k, v_x.off)] += mr * du_dx[j];
729 }
730 if (v_x.present) B.Dd[M(k, v_x.off)] += QY_D * exd * qspec_base[k] * dx;
731 }
732 }
733 const BayesianDecayVariableInfo v_qd = VI("QY_D");
734 if (v_qd.present) for (std::size_t k = 0; k < K; ++k) B.Dd[M(k, v_qd.off)] += exd * q[k] * bayesian_decay_dxdz(v_qd, QY_D);
735 if (exs == "EX_DR") { const BayesianDecayVariableInfo v_ed = VI("EX_DR");
736 if (v_ed.present) for (std::size_t k = 0; k < K; ++k) B.Dd[M(k, v_ed.off)] += QY_D * q[k] * bayesian_decay_dxdz(v_ed, EX_DR); }
737 }
738 // --- acceptor ---
739 if (sc.has_acc) {
740 std::string ds = sc.direct;
741 const bool has_n = ds.find("n_da") != std::string::npos, has_pm = ds.find("per_mol") != std::string::npos;
742 const bool has_ex = ds.find("EX_AG") != std::string::npos;
743 const double direct = ds.empty() ? 0.0 : (has_ex ? EX_AG : 1.0) * (has_n ? n_da : 1.0) * (has_pm ? per_mol : 1.0);
744 const double d_direct_dspec = has_pm ? (has_ex ? EX_AG : 1.0) * (has_n ? n_da : 1.0) : 0.0;
745 const double d_direct_du = has_n ? (has_ex ? EX_AG : 1.0) * (has_pm ? per_mol : 1.0) : 0.0;
746 const double d_direct_dex = has_ex ? (has_n ? n_da : 1.0) * (has_pm ? per_mol : 1.0) : 0.0;
747 std::vector<double> qa(K, 0.0), qar(K, 0.0);
748 if (sens) for (std::size_t j = 0; j < nR; ++j) { if (u[j] == 0.0) continue;
749 for (std::size_t l = 0; l < nA; ++l) for (std::size_t k = 0; k < K; ++k) {
750 qa[k] += u[j] * w_a[l] * MA[(j * nA + l) * K + k];
751 for (std::size_t r = 0; r < nra; ++r) qar[k] += w_rho_a[r] * u[j] * w_a[l] * MAr[((r * nR + j) * nA + l) * K + k];
752 } }
753 for (std::size_t k = 0; k < K; ++k) { B.acc[k] = QY_A * (qa[k] + a_dir[k] * direct); B.accr[k] = QY_A * (qar[k] + a_dir_r[k] * direct); }
754 // spectrum: sensitised matrices Qa, Qar (K, Kint) and the direct term's per_mol
755 if (v_eps.present) {
756 std::vector<double> Qa(K * Kint, 0.0), Qar(K * Kint, 0.0);
757 if (sens) for (std::size_t j = 0; j < nR; ++j) { if (u[j] == 0.0) continue;
758 for (std::size_t l = 0; l < nA; ++l) { // saxpy per block, r outside
759 const double ujl = u[j] * w_a[l];
760 const double* M0 = e.S_Ag->d.data() + (j * nA + l) * K * Kint;
761 for (std::size_t kk = 0; kk < K * Kint; ++kk) Qa[kk] += ujl * M0[kk];
762 for (std::size_t r = 0; r < nra; ++r) {
763 const double wr = w_rho_a[r] * ujl;
764 const double* Mr = e.S_Ag_rot_r->d.data() + ((r * nR + j) * nA + l) * K * Kint;
765 for (std::size_t kk = 0; kk < K * Kint; ++kk) Qar[kk] += wr * Mr[kk];
766 }
767 } }
768 for (std::size_t k = 0; k < K; ++k) for (std::size_t i = 0; i < Kint; ++i) {
769 const double ma = QY_A * (Qa[k * Kint + i] + a_dir[k] * d_direct_dspec);
770 const double mr = QY_A * (Qar[k * Kint + i] + a_dir_r[k] * d_direct_dspec);
771 for (std::size_t c = v_eps.off; c < v_eps.off + v_eps.size; ++c) {
772 B.Da[M(k, c)] += ma * dspec[i * dim + c]; B.Dr[M(k, c)] += mr * dspec[i * dim + c];
773 }
774 }
775 }
776 // u (through c and x)
777 if (fret) for (std::size_t k = 0; k < K; ++k) for (std::size_t j = 0; j < nR; ++j) {
778 double dqa = a_dir[k] * d_direct_du, dqr = a_dir_r[k] * d_direct_du;
779 if (sens) for (std::size_t l = 0; l < nA; ++l) {
780 dqa += w_a[l] * MA[(j * nA + l) * K + k];
781 for (std::size_t r = 0; r < nra; ++r) dqr += w_rho_a[r] * w_a[l] * MAr[((r * nR + j) * nA + l) * K + k];
782 }
783 dqa *= QY_A; dqr *= QY_A;
784 for (std::size_t b = 0; b < nz; ++b) {
785 const double dudc = (1.0 - x) * dp[j * dim + v_c.off + b];
786 B.Da[M(k, v_c.off + b)] += dqa * dudc; B.Dr[M(k, v_c.off + b)] += dqr * dudc;
787 }
788 if (v_x.present) { B.Da[M(k, v_x.off)] += dqa * du_dx[j]; B.Dr[M(k, v_x.off)] += dqr * du_dx[j]; }
789 }
790 // w_a
791 if (v_wa.present) {
792 const std::size_t nw = nA - 1;
793 for (std::size_t k = 0; k < K; ++k) for (std::size_t l = 0; l < nA; ++l) {

892 }

894 for (std::size_t k = 0; k < K; ++k) {

898 if (v_gc.present) Jk[M(k, v_gc.off)] += mix[k] / Gc * bayesian_decay_dxdz(v_gc, Gc);
899 }

910 for (std::size_t k = 0; k < K; ++k) {

913 if (v_bk.present) J[(ic * K + k) * dim + v_bk.off] += dpar[k * 4 + 3] * bayesian_decay_dxdz(v_bk, ip.background);
914 }
915 });

920// the response with its tangents, the expected counts and the count Jacobian
921//
922// Forward-mode by hand through the four steps of InstrumentModel.basis --
923// drop_background, the clamped phase-ramp shift, the unit-sum normalisation,
924// the exact periodic kernel with its clamp and column normalisation -- for
925// the response's own coordinates: its background, the detector's shift and
926// the sample's time offset. Every step is linear in its tangent apart from
927// the softplus (torch's threshold 20 in its derivative too) and the clamps
928// (derivative 1 where the argument is >= 0, torch's clamp_min rule).
929// ---------------------------------------------------------------------------

931struct BayesianDecayResponseTangents {
932 std::vector<double> B; // (n, K)
933 std::vector<std::vector<double>> dB; // per tangent, (n, K), d/dz
934 std::vector<std::string> names; // the free coordinates the tangents belong to
935 std::vector<std::size_t> offsets; // their offsets in theta
936 std::vector<double> chain; // d constrained / d z
937};

941 if (v.count(name)) return v.at(name)[0];

947inline BayesianDecayResponseTangents bayesian_decay_response_basis(const BayesianDecayExperiment& f, const std::vector<double>& tau_c, const BayesianDecayValues& v,
948 const std::string& det, const std::string& samp, bool derivs = true,
949 bool clamp_in_tangent = true) {
950 (void)tau_c; // the kernel on this grid is the experiment's own (f.kernel())
951 const BayesianDecayAxis ax = f.axis();
952
953 const double soft = f.soft, dt = ax.dt;
954 std::string phys; double off_ns = 0.0;
955 for (auto& pt : f.parts) if (pt.det == det) { phys = pt.phys; off_ns = pt.offset_ns; break; }
956 std::size_t ri = 0;
957 for (; ri < f.responses.size(); ++ri)
958 if (f.responses[ri].sample == samp && f.responses[ri].det == phys) break;
959 const BayesianDecayResponse& r = f.responses[ri];
960 const std::string bg_n = r.bg_var, sh_n = r.shift_var, of_n = r.offset_var;
961 const std::vector<double>& raw = f["response_" + std::to_string(ri)].d;
962
963 BayesianDecayResponseTangents out;
964 //: each free coordinate's tangent is d/d b (the background fraction) or d/d shift,
965 //: times its transform's chain factor; the detector's shift and the sample's
966 //: offset share the shift tangent
967 std::vector<int> kind_of; // per name: 0 background, 1 shift, 2 tail fraction, 3 tail log10 tau
968 auto add = [&](const std::string& nm, int k) {
969 if (nm.empty() || !derivs) return;
970 const BayesianDecayVariableInfo vi = bayesian_decay_variable_info(f, nm);
971 if (!vi.present) return;
972 out.names.push_back(nm); out.offsets.push_back(vi.off); kind_of.push_back(k);
973 out.chain.push_back(bayesian_decay_dxdz(vi, bayesian_decay_constrained_scalar(f, v, nm)));
974 };
975 add(bg_n, 0); add(sh_n, 1); add(of_n, 1);
976 const std::string ta_n = r.tail_fraction_var, tt_n = r.tail_log10_tau_var;
977 if ((ta_n.empty()) != (tt_n.empty()))
978 throw std::runtime_error("bayesian_decay_response_basis: a response tail needs both tail_fraction_var and tail_log10_tau_var");
979 add(ta_n, 2); add(tt_n, 3);
980 const double shift_ns = bayesian_decay_constrained_scalar(f, v, sh_n) + off_ns + (of_n.empty() ? 0.0 : bayesian_decay_constrained_scalar(f, v, of_n));
981 BayesianResponseOptions opt;
982 opt.soft = soft;
983 opt.clamp_in_tangent = clamp_in_tangent;
984 opt.floor = f.response_floor;
985 if (!ta_n.empty()) {
986 opt.tail_fraction = bayesian_decay_constrained_scalar(f, v, ta_n);
987 opt.tail_log10_tau = bayesian_decay_constrained_scalar(f, v, tt_n);
988 opt.tail_tangents = derivs;
989 }
990 BayesianResponseBasis rb = bayesian_response_basis(f.kernel(), raw, bayesian_decay_constrained_scalar(f, v, bg_n),
991 shift_ns / dt, derivs && !out.names.empty(), opt);
992 out.dB.resize(out.names.size());
993 for (std::size_t t = 0; t < out.names.size(); ++t) {
994 const int kd = kind_of[t];
995 out.dB[t] = kd == 0 ? rb.dB_background : kd == 1 ? rb.dB_shift : kd == 2 ? rb.dB_tail_fraction : rb.dB_tail_log10_tau;
996 const double c = out.chain[t] * (kd == 1 ? 1.0 / dt : 1.0);
997 for (double& q : out.dB[t]) q *= c;
998 }
999 out.B = std::move(rb.B);
1000 return out;
1001}

1088 for (std::size_t q = 0; q < parts.size(); ++q) {
1089 const BayesianDecayPart& pt = parts[q];
1090 const std::size_t o = pt.out, ai = pt.amp_index;

1102 if (v_bk.present) jrow[v_bk.off] += total * ((1.0 + scat) / (den * den)) * bayesian_decay_dxdz(v_bk, bkg);

1116 }
