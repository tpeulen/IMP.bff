180inline BayesianResponseBasis bayesian_response_basis(const BayesianPeriodicKernel& kernel, const std::vector<double>& measured,
181 double background_fraction, double shift_bins, bool tangents,
182 const BayesianResponseOptions& opt = BayesianResponseOptions()) {
183 using cd = std::complex<double>;
184 const BayesianDecayAxis& ax = kernel.axis();
185 const std::size_t n = measured.size(), np = ax.n_period, h = np / 2 + 1, nt = kernel.tau().size(), K = nt + 2;
186 const double two_pi = 2.0 * 3.14159265358979323846;
187 BayesianResponseBasis out;
188 out.n = n; out.K = K;
189
190 // 1. the background through internal/ResponseFunction.h (mode fraction_of_support), and d/d b
191 internal::ResponsePreparation prep;
192 prep.active = opt.remove_background;
193 prep.background_mode = internal::ResponseBackground::fraction_of_support;
194 prep.background = background_fraction;
195 prep.soft = opt.soft;
196 std::vector<double> cleaned, du_b(tangents ? n : 0, 0.0);
197 const double* cleaned_ptr = internal::clean_response(measured, prep, cleaned);
198 std::vector<double> u(cleaned_ptr, cleaned_ptr + n);
199 if (tangents) internal::clean_response_background_derivative(measured, prep, du_b.data());
200 // 2. the shift, clamped; tangents in b (through the ramp) and in the shift
201 std::vector<cd> U(n / 2 + 1), T(n / 2 + 1), ramp(n / 2 + 1);
202 bayesian_rfft(u.data(), n, U.data());
203 for (std::size_t k = 0; k <= n / 2; ++k) ramp[k] = std::exp(cd(0.0, -two_pi * double(k) / double(n) * shift_bins));
204 for (std::size_t k = 0; k <= n / 2; ++k) T[k] = U[k] * ramp[k];
205 std::vector<double> y(n), dy_b(tangents ? n : 0), dy_s(tangents ? n : 0);
206 bayesian_irfft(T.data(), n, y.data());
207 const BayesianResponseFloor fmode = opt.clamp ? opt.floor : BayesianResponseFloor::none;
208 if (tangents) {
209 bayesian_rfft(du_b.data(), n, T.data());
210 for (std::size_t k = 0; k <= n / 2; ++k) T[k] *= ramp[k];
211 bayesian_irfft(T.data(), n, dy_b.data());
212 for (std::size_t k = 0; k <= n / 2; ++k) T[k] = U[k] * ramp[k] * cd(0.0, -two_pi * double(k) / double(n));
213 bayesian_irfft(T.data(), n, dy_s.data());
214 if (fmode == BayesianResponseFloor::hard && opt.clamp_in_tangent)
215 for (std::size_t i = 0; i < n; ++i) if (!(y[i] >= 0.0)) { dy_b[i] = 0.0; dy_s[i] = 0.0; }
216 }
217 if (fmode == BayesianResponseFloor::hard) for (double& t : y) t = std::max(t, 0.0);
218 // 3. unit sum, through internal/ResponseFunction.h
219 internal::normalize_unit_sum(y.data(), n, tangents ? std::vector<double*>{dy_b.data(), dy_s.data()} : std::vector<double*>{}, 0.0);
220 out.response = y;
221 // 4. the basis and its tangents
222 out.B.assign(n * K, 0.0);
223 if (tangents) { out.dB_background.assign(n * K, 0.0); out.dB_shift.assign(n * K, 0.0); }
224 for (std::size_t i = 0; i < n; ++i) {
225 out.B[i * K] = y[i]; out.B[i * K + K - 1] = 1.0 / double(n);
226 if (tangents) { out.dB_background[i * K] = dy_b[i]; out.dB_shift[i * K] = dy_s[i]; }
227 }
228 std::vector<double> buf(np, 0.0), col(np), dcol(np);
229 std::vector<cd> Y(h), Db(tangents ? h : 0), Ds(tangents ? h : 0), tmp(h);
230 auto to_period = [&](const std::vector<double>& v, std::vector<cd>& V) {
231 std::fill(buf.begin(), buf.end(), 0.0);
232 for (std::size_t i = 0; i < std::min(np, n); ++i) buf[i] = v[i];
233 bayesian_rfft(buf.data(), np, V.data());
234 };
235 to_period(y, Y);
236 if (tangents) { to_period(dy_b, Db); to_period(dy_s, Ds); }
237 //: PRD-144, the response tail: in the lifetime columns the response is (1 - a) u + a u (*) k(tau),
238 //: i.e. its transform times M = (1 - a) + a KT. d/da multiplies by KT - 1; d/d log10 tau by
239 //: a ln10 (KT1 - KT), since d k0 / d ln tau = k1 - k0 for the bin-integrated periodic kernels.
240 const bool tail = opt.tail_fraction != 0.0 || (tangents && opt.tail_tangents);
241 const bool tail_t = tangents && opt.tail_tangents;
242 std::vector<cd> Da(tail_t ? h : 0), Dt(tail_t ? h : 0);
243 if (tail) {
244 const double a = opt.tail_fraction, tau = std::pow(10.0, opt.tail_log10_tau), ln10 = std::log(10.0);
245 std::vector<double> k0(np), k1(np);
246 periodic_decay_kernel(k0.data(), int(np), ax.dt, tau, 0);
247 periodic_decay_kernel(k1.data(), int(np), ax.dt, tau, 1);
248 std::vector<cd> KT(h), KT1(h);
249 bayesian_rfft(k0.data(), np, KT.data());
250 bayesian_rfft(k1.data(), np, KT1.data());
251 for (std::size_t k = 0; k < h; ++k) {
252 const cd M = (1.0 - a) + a * KT[k];
253 if (tail_t) { Da[k] = Y[k] * (KT[k] - 1.0); Dt[k] = Y[k] * (a * ln10) * (KT1[k] - KT[k]); }
254 Y[k] *= M;
255 if (tangents) { Db[k] *= M; Ds[k] *= M; }
256 }
257 }
258 if (tail_t) { out.dB_tail_fraction.assign(n * K, 0.0); out.dB_tail_log10_tau.assign(n * K, 0.0); }
259 for (std::size_t c = 0; c < nt; ++c) {
260 const std::vector<cd>& KF = kernel.fft(c);
261 for (std::size_t k = 0; k < h; ++k) tmp[k] = KF[k] * Y[k];
262 bayesian_irfft(tmp.data(), np, col.data());
263 //: clamp at zero, then unit sum with the tangents masked by the clamp -- through
264 //: internal/ResponseFunction.h's normalize_unit_sum
265 std::vector<double> cc(n), d_b(tangents ? n : 0), d_s(tangents ? n : 0), d_a(tail_t ? n : 0), d_t(tail_t ? n : 0);
266 const bool col_floor = opt.floor == BayesianResponseFloor::hard;
267 for (std::size_t i = 0; i < n; ++i) cc[i] = col_floor ? std::max(col[i], 0.0) : col[i];
268 if (tangents) {
269 const int n_which = tail_t ? 4 : 2;
270 for (int which = 0; which < n_which; ++which) {
271 const std::vector<cd>& D = which == 0 ? Db : which == 1 ? Ds : which == 2 ? Da : Dt;
272 std::vector<double>& d = which == 0 ? d_b : which == 1 ? d_s : which == 2 ? d_a : d_t;
273 for (std::size_t k = 0; k < h; ++k) tmp[k] = KF[k] * D[k];
274 bayesian_irfft(tmp.data(), np, dcol.data());
275 for (std::size_t i = 0; i < n; ++i) d[i] = (!col_floor || col[i] >= 0.0) ? dcol[i] : 0.0;
276 }
277 }
278 std::vector<double*> tg;
279 if (tangents) { tg.push_back(d_b.data()); tg.push_back(d_s.data()); }
280 if (tail_t) { tg.push_back(d_a.data()); tg.push_back(d_t.data()); }
281 internal::normalize_unit_sum(cc.data(), n, tg);
282 for (std::size_t i = 0; i < n; ++i) {
283 out.B[i * K + 1 + c] = cc[i];
284 if (tangents) { out.dB_background[i * K + 1 + c] = d_b[i]; out.dB_shift[i * K + 1 + c] = d_s[i]; }
285 if (tail_t) { out.dB_tail_fraction[i * K + 1 + c] = d_a[i]; out.dB_tail_log10_tau[i * K + 1 + c] = d_t[i]; }
286 }
287 }
288 return out;
289}
