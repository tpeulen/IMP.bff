64inline RunsTest runs_test(const double* x, std::size_t n, double cutoff = 0.0,
65 bool continuity_correction = false) {
66 RunsTest r;
67 if (n == 0) return r;
68 std::vector<bool> above(n);
69 for (std::size_t i = 0; i < n; ++i) {
70 above[i] = (x[i] >= cutoff);
71 if (above[i]) ++r.n_above; else ++r.n_below;
72 }
73 r.n_runs = 1;
74 for (std::size_t i = 1; i < n; ++i) if (above[i] != above[i - 1]) ++r.n_runs;
75 const double n1 = double(r.n_above), n2 = double(r.n_below), N = n1 + n2;
76 if (n1 == 0.0 || n2 == 0.0 || N < 2.0) { r.z = 0.0; r.p_value = 1.0; return r; }
77 const double mean = 2.0 * n1 * n2 / N + 1.0;
78 const double var = 2.0 * n1 * n2 * (2.0 * n1 * n2 - N) / (N * N * (N - 1.0));
79 if (!(var > 0.0)) { r.z = 0.0; r.p_value = 1.0; return r; }
80 double diff = double(r.n_runs) - mean;
81 if (continuity_correction) diff -= (diff > 0.0 ? 0.5 : -0.5);
82 r.z = diff / std::sqrt(var);
83 r.p_value = std::erfc(std::fabs(r.z) / std::sqrt(2.0));
84 return r;
85}

132inline PoissonGoodnessOfFit poisson_goodness_of_fit(const std::vector<double>& y, const std::vector<double>& m,
133 const std::vector<double>& mask, std::size_t n_hist,
134 double n_params, int n_draws = 150, std::uint64_t seed = 0) {
135 const std::size_t n = n_hist ? y.size() / n_hist : 0;
136 PoissonGoodnessOfFit G;
137 G.histograms.resize(n_hist);
138 std::vector<std::vector<std::size_t>> sel(n_hist);
139 double nsel_all = 0.0;
140 for (std::size_t k = 0; k < n_hist; ++k) {
141 PoissonHistogramFit& h = G.histograms[k];
142 std::vector<double> ys, ms, signs;
143 for (std::size_t i = 0; i < n; ++i) {
144 if (!(mask[k * n + i] > 0.0)) continue;
145 sel[k].push_back(i);
146 const double l = std::max(m[k * n + i], 1e-12), yy = y[k * n + i];
147 ys.push_back(yy); ms.push_back(l);
148 const double w = (yy - l) / std::sqrt(l);
149 h.weighted_residuals.push_back(w);
150 signs.push_back(w > 0.0 ? 1.0 : (w < 0.0 ? -1.0 : 0.0));
151 }
152 h.deviance = poisson_deviance(ys.data(), ms.data(), ys.size());
153 h.dof = std::max(double(sel[k].size()) - n_params / double(std::max<std::size_t>(n_hist, 1)), 1.0);
154 h.deviance_per_dof = h.deviance / h.dof;
155 double mean_sign = 0.0;
156 for (double t : signs) mean_sign += t;
157 if (!signs.empty()) mean_sign /= double(signs.size());
158 h.runs_p = runs_test(signs.data(), signs.size(), mean_sign).p_value;
159 G.deviance += h.deviance;
160 nsel_all += double(sel[k].size());
161 }
162 G.dof = std::max(nsel_all - n_params, 1.0);
163 G.deviance_per_dof = G.deviance / G.dof;
164 std::mt19937_64 rng(seed);
165 std::vector<std::vector<double>> per(n_hist);
166 std::vector<double> tot, yd, md;
167 for (int d = 0; d < n_draws; ++d) {
168 double all = 0.0;
169 for (std::size_t k = 0; k < n_hist; ++k) {
170 yd.clear(); md.clear();
171 for (std::size_t i : sel[k]) {
172 const double mu = std::max(m[k * n + i], 1e-12);
173 yd.push_back(double(IMP::bff::internal::portable::poisson_distribution<long long>(mu)(rng)));
174 md.push_back(mu);
175 }
176 const double dk = poisson_deviance(yd.data(), md.data(), yd.size());
177 per[k].push_back(dk / G.histograms[k].dof);
178 all += dk;
179 }
180 tot.push_back(all / G.dof);
181 }
182 auto moments = [](const std::vector<double>& v, double& mean, double& sd) {
183 mean = 0.0; sd = 0.0;
184 if (v.empty()) return;
185 for (double t : v) mean += t;
186 mean /= double(v.size());
187 for (double t : v) sd += (t - mean) * (t - mean);
188 sd = std::sqrt(sd / double(v.size()));
189 };
190 for (std::size_t k = 0; k < n_hist; ++k) {
191 PoissonHistogramFit& h = G.histograms[k];
192 moments(per[k], h.reference_mean, h.reference_sd);
193 h.z = (h.deviance_per_dof - h.reference_mean) / std::max(h.reference_sd, 1e-12);
194 }
195 moments(tot, G.reference_mean, G.reference_sd);
196 G.z = (G.deviance_per_dof - G.reference_mean) / std::max(G.reference_sd, 1e-12);
197 return G;
198}
