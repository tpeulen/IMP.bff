62inline RunsTest runs_test(const double* x, std::size_t n, double cutoff = 0.0,
63 bool continuity_correction = false) {
64 RunsTest r;
65 if (n == 0) return r;
66 std::vector<bool> above(n);
67 for (std::size_t i = 0; i < n; ++i) {
68 above[i] = (x[i] >= cutoff);
69 if (above[i]) ++r.n_above; else ++r.n_below;
70 }
71 r.n_runs = 1;
72 for (std::size_t i = 1; i < n; ++i) if (above[i] != above[i - 1]) ++r.n_runs;
73 const double n1 = double(r.n_above), n2 = double(r.n_below), N = n1 + n2;
74 if (n1 == 0.0 || n2 == 0.0 || N < 2.0) { r.z = 0.0; r.p_value = 1.0; return r; }
75 const double mean = 2.0 * n1 * n2 / N + 1.0;
76 const double var = 2.0 * n1 * n2 * (2.0 * n1 * n2 - N) / (N * N * (N - 1.0));
77 if (!(var > 0.0)) { r.z = 0.0; r.p_value = 1.0; return r; }
78 double diff = double(r.n_runs) - mean;
79 if (continuity_correction) diff -= (diff > 0.0 ? 0.5 : -0.5);
80 r.z = diff / std::sqrt(var);
81 r.p_value = std::erfc(std::fabs(r.z) / std::sqrt(2.0));
82 return r;
83}

121inline PoissonGoodnessOfFit poisson_goodness_of_fit(const std::vector<double>& y, const std::vector<double>& m,
122 const std::vector<double>& mask, std::size_t n_hist,
123 double n_params, int n_draws = 150, std::uint64_t seed = 0) {
124 const std::size_t n = n_hist ? y.size() / n_hist : 0;
125 PoissonGoodnessOfFit G;
126 G.histograms.resize(n_hist);
127 std::vector<std::vector<std::size_t>> sel(n_hist);
128 double nsel_all = 0.0;
129 for (std::size_t k = 0; k < n_hist; ++k) {
130 PoissonHistogramFit& h = G.histograms[k];
131 std::vector<double> ys, ms, signs;
132 for (std::size_t i = 0; i < n; ++i) {
133 if (!(mask[k * n + i] > 0.0)) continue;
134 sel[k].push_back(i);
135 const double l = std::max(m[k * n + i], 1e-12), yy = y[k * n + i];
136 ys.push_back(yy); ms.push_back(l);
137 const double w = (yy - l) / std::sqrt(l);
138 h.weighted_residuals.push_back(w);
139 signs.push_back(w > 0.0 ? 1.0 : (w < 0.0 ? -1.0 : 0.0));
140 }
141 h.deviance = poisson_deviance(ys.data(), ms.data(), ys.size());
142 h.dof = std::max(double(sel[k].size()) - n_params / double(std::max<std::size_t>(n_hist, 1)), 1.0);
143 h.deviance_per_dof = h.deviance / h.dof;
144 double mean_sign = 0.0;
145 for (double t : signs) mean_sign += t;
146 if (!signs.empty()) mean_sign /= double(signs.size());
147 h.runs_p = runs_test(signs.data(), signs.size(), mean_sign).p_value;
148 G.deviance += h.deviance;
149 nsel_all += double(sel[k].size());
150 }
151 G.dof = std::max(nsel_all - n_params, 1.0);
152 G.deviance_per_dof = G.deviance / G.dof;
153 std::mt19937_64 rng(seed);
154 std::vector<std::vector<double>> per(n_hist);
155 std::vector<double> tot, yd, md;
156 for (int d = 0; d < n_draws; ++d) {
157 double all = 0.0;
158 for (std::size_t k = 0; k < n_hist; ++k) {
159 yd.clear(); md.clear();
160 for (std::size_t i : sel[k]) {
161 const double mu = std::max(m[k * n + i], 1e-12);
162 yd.push_back(double(std::poisson_distribution<long long>(mu)(rng)));
163 md.push_back(mu);
164 }
165 const double dk = poisson_deviance(yd.data(), md.data(), yd.size());
166 per[k].push_back(dk / G.histograms[k].dof);
167 all += dk;
168 }
169 tot.push_back(all / G.dof);
170 }
171 auto moments = [](const std::vector<double>& v, double& mean, double& sd) {
172 mean = 0.0; sd = 0.0;
173 if (v.empty()) return;
174 for (double t : v) mean += t;
175 mean /= double(v.size());
176 for (double t : v) sd += (t - mean) * (t - mean);
177 sd = std::sqrt(sd / double(v.size()));
178 };
179 for (std::size_t k = 0; k < n_hist; ++k) {
180 PoissonHistogramFit& h = G.histograms[k];
181 moments(per[k], h.reference_mean, h.reference_sd);
182 h.z = (h.deviance_per_dof - h.reference_mean) / std::max(h.reference_sd, 1e-12);
183 }
184 moments(tot, G.reference_mean, G.reference_sd);
185 G.z = (G.deviance_per_dof - G.reference_mean) / std::max(G.reference_sd, 1e-12);
186 return G;
187}
