89inline void check(const McmcChains& x) {
90 if (x.empty() || x[0].empty()) throw std::invalid_argument("mcmc diagnostics: no draws");
91 for (const auto& c : x)
92 if (c.size() != x[0].size()) throw std::invalid_argument("mcmc diagnostics: chains of unequal length");

96inline McmcChains split(const McmcChains& x) {
97 const std::size_t n = x[0].size(), h = n / 2;
98 McmcChains out;
99 out.reserve(2 * x.size());
100 for (const auto& c : x) out.emplace_back(c.begin(), c.begin() + std::ptrdiff_t(h));
101 for (const auto& c : x) out.emplace_back(c.end() - std::ptrdiff_t(h), c.end());
102 return out;
103}

105inline std::vector<double> pooled(const McmcChains& x) {
106 std::vector<double> v;
107 for (const auto& c : x) v.insert(v.end(), c.begin(), c.end());
108 return v;

112inline double median(const McmcChains& x) {
113 std::vector<double> v = pooled(x);
114 std::sort(v.begin(), v.end());
115 const std::size_t n = v.size();
116 return n % 2 ? v[n / 2] : 0.5 * (v[n / 2 - 1] + v[n / 2]);

120inline double quantile(const McmcChains& x, double prob) {
121 std::vector<double> v = pooled(x);
122 std::sort(v.begin(), v.end());
123 const double h = (double(v.size()) - 1.0) * prob;
124 const std::size_t lo = std::size_t(std::floor(h));
125 if (lo + 1 >= v.size()) return v.back();
126 return v[lo] + (h - double(lo)) * (v[lo + 1] - v[lo]);
127}

130inline McmcChains z_scale(const McmcChains& x) {
131 const std::size_t m = x.size(), n = x[0].size(), N = m * n;
132 std::vector<double> v = pooled(x);
133 std::vector<std::size_t> idx(N);
134 std::iota(idx.begin(), idx.end(), std::size_t(0));
135 std::stable_sort(idx.begin(), idx.end(), [&](std::size_t a, std::size_t b) { return v[a] < v[b]; });
136 std::vector<double> rank(N);
137 for (std::size_t i = 0; i < N;) {
138 std::size_t j = i;
139 while (j + 1 < N && v[idx[j + 1]] == v[idx[i]]) ++j;
140 const double r = 0.5 * (double(i + 1) + double(j + 1));
141 for (std::size_t k = i; k <= j; ++k) rank[idx[k]] = r;
142 i = j + 1;
143 }
144 const double c = 3.0 / 8.0;
145 McmcChains out(m, std::vector<double>(n));
146 for (std::size_t a = 0; a < m; ++a)
147 for (std::size_t i = 0; i < n; ++i) out[a][i] = normal_quantile((rank[a * n + i] - c) / (double(N) - 2.0 * c + 1.0));
148 return out;
149}

152inline double rhat(const McmcChains& x) {
153 const std::size_t m = x.size(), n = x[0].size();
154 std::vector<double> mean(m, 0.0), var(m, 0.0);
155 for (std::size_t a = 0; a < m; ++a) {
156 for (double v : x[a]) mean[a] += v;
157 mean[a] /= double(n);
158 for (double v : x[a]) var[a] += (v - mean[a]) * (v - mean[a]);
159 var[a] /= double(n - 1);
160 }
161 double mm = 0.0;
162 for (double v : mean) mm += v;
163 mm /= double(m);
164 double vm = 0.0;
165 for (double v : mean) vm += (v - mm) * (v - mm);
166 vm /= double(m - 1);
167 const double B = double(n) * vm;
168 double W = 0.0;
169 for (double v : var) W += v;
170 W /= double(m);
171 return std::sqrt((B / W + double(n) - 1.0) / double(n));
172}

175inline double ess(const McmcChains& x) {
176 const std::size_t m = x.size(), n = x[0].size();
177 double lo = std::numeric_limits<double>::infinity(), hi = -lo;
178 for (const auto& c : x) for (double v : c) { lo = std::min(lo, v); hi = std::max(hi, v); }
179 if (hi - lo < 1e-15) return double(m * n);
180 std::vector<std::vector<double>> centred(m, std::vector<double>(n));
181 std::vector<double> chain_mean(m, 0.0);
182 for (std::size_t a = 0; a < m; ++a) {
183 for (double v : x[a]) chain_mean[a] += v;
184 chain_mean[a] /= double(n);
185 for (std::size_t i = 0; i < n; ++i) centred[a][i] = x[a][i] - chain_mean[a];
186 }
187 std::vector<double> mean_acov; //: mean over chains of the lag-t autocovariance, filled on demand
188 auto acov = [&](std::size_t t) {
189 while (mean_acov.size() <= t) {
190 const std::size_t k = mean_acov.size();
191 double s = 0.0;
192 for (std::size_t a = 0; a < m; ++a) {
193 double sa = 0.0;
194 for (std::size_t i = 0; i + k < n; ++i) sa += centred[a][i] * centred[a][i + k];
195 s += sa / double(n);
196 }
197 mean_acov.push_back(s / double(m));
198 }
199 return mean_acov[t];
200 };
201 const double mean_var = acov(0) * double(n) / (double(n) - 1.0);
202 double var_plus = mean_var * (double(n) - 1.0) / double(n);
203 if (m > 1) {
204 double mm = 0.0;
205 for (double v : chain_mean) mm += v;
206 mm /= double(m);
207 double vm = 0.0;
208 for (double v : chain_mean) vm += (v - mm) * (v - mm);
209 var_plus += vm / double(m - 1);
210 }
211 std::vector<double> rho(std::max<std::size_t>(n, 2), 0.0);
212 double rho_even = 1.0;
213 rho[0] = rho_even;
214 double rho_odd = 1.0 - (mean_var - (n > 1 ? acov(1) : 0.0)) / var_plus;
215 rho[1] = rho_odd;
216 std::size_t t = 1;
217 //: signed: with fewer than 4 draws per (split) chain arviz's `t < n - 3` is false from the start,
218 //: and an unsigned n - 3 would wrap around
219 while (std::ptrdiff_t(t) < std::ptrdiff_t(n) - 3 && (rho_even + rho_odd) > 0.0) { //: Geyer's initial positive sequence
220 rho_even = 1.0 - (mean_var - acov(t + 1)) / var_plus;
221 rho_odd = 1.0 - (mean_var - acov(t + 2)) / var_plus;
222 if (rho_even + rho_odd >= 0.0) { rho[t + 1] = rho_even; rho[t + 2] = rho_odd; }
223 t += 2;
224 }
225 const std::ptrdiff_t max_t = std::ptrdiff_t(t) - 2;
226 if (rho_even > 0.0) rho[std::size_t(max_t + 1)] = rho_even;
227 for (std::ptrdiff_t u = 1; u <= max_t - 2; u += 2) { //: Geyer's initial monotone sequence
228 if (rho[std::size_t(u + 1)] + rho[std::size_t(u + 2)] > rho[std::size_t(u - 1)] + rho[std::size_t(u)]) {
229 rho[std::size_t(u + 1)] = 0.5 * (rho[std::size_t(u - 1)] + rho[std::size_t(u)]);
230 rho[std::size_t(u + 2)] = rho[std::size_t(u + 1)];
231 }
232 }
233 const double total = double(m * n);
234 double tau = -1.0;
235 for (std::ptrdiff_t u = 0; u <= max_t; ++u) tau += 2.0 * rho[std::size_t(u)];
236 if (max_t + 1 < std::ptrdiff_t(n)) tau += rho[std::size_t(max_t + 1)];
237 tau = std::max(tau, 1.0 / std::log10(total));
238 return total / tau;
239}

244inline double rhat_rank(const McmcChains& x) {
245 mcmc_detail::check(x);
246 if (x.size() < 2 || x[0].size() < 4) return std::numeric_limits<double>::quiet_NaN();
247 const McmcChains s = mcmc_detail::split(x);
248 const double bulk = mcmc_detail::rhat(mcmc_detail::z_scale(s));
249 const double med = mcmc_detail::median(s);
250 McmcChains f = s;
251 for (auto& c : f) for (double& v : c) v = std::fabs(v - med);
252 const double tail = mcmc_detail::rhat(mcmc_detail::z_scale(f));
253 return std::max(bulk, tail);
254}

264inline double ess_quantile(const McmcChains& x, double prob) {
265 mcmc_detail::check(x);
266 if (x[0].size() < 4) return std::numeric_limits<double>::quiet_NaN();
267 const double q = mcmc_detail::quantile(x, prob);
268 McmcChains ind = x;
269 for (auto& c : ind) for (double& v : c) v = v <= q ? 1.0 : 0.0;
270 return mcmc_detail::ess(mcmc_detail::split(ind));
271}

274inline double ess_tail(const McmcChains& x, double prob = 0.05) {
275 const double p = std::min(prob, 1.0 - prob);
276 return std::min(ess_quantile(x, p), ess_quantile(x, 1.0 - p));
277}

287inline double mcse_mean(const McmcChains& x) {
288 const std::vector<double> v = mcmc_detail::pooled(x);
289 double mean = 0.0;
290 for (double a : v) mean += a;
291 mean /= double(v.size());
292 double s2 = 0.0;
293 for (double a : v) s2 += (a - mean) * (a - mean);
294 s2 /= double(v.size() - 1);
295 return std::sqrt(s2) / std::sqrt(ess_mean(x));
296}

305inline ChainSummary summarize_chains(const McmcChains& chains) {
306 mcmc_detail::check(chains);
307 ChainSummary c;
308 std::size_t N = 0;
309 for (const auto& ch : chains)
310 for (double v : ch) { c.mean += v; ++N; }
311 c.mean /= double(N);
312 for (const auto& ch : chains)
313 for (double v : ch) c.sd += (v - c.mean) * (v - c.mean);
314 c.sd = std::sqrt(c.sd / double(N - 1));
315 c.rhat = rhat_rank(chains);
316 c.ess_bulk = ess_bulk(chains);
317 c.ess_tail = ess_tail(chains);
318 c.mcse = mcse_mean(chains);
319 c.tau = double(N) / ess_mean(chains);
320 return c;
321}
