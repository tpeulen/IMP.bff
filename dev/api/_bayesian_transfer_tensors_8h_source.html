41inline std::vector<double> bayesian_log_lifetime_grid(double log10_lo, double log10_hi, int per_decade) {
42 const double step = 1.0 / double(per_decade);
43 std::vector<double> tau;
44 for (int k = 1;; ++k) {
45 const double u = log10_lo + k * step;
46 if (u > log10_hi - step + 1e-9) break;
47 tau.push_back(std::pow(10.0, u));
48 }
49 return tau;
50}

61inline std::vector<double> bayesian_gaussian_response(std::size_t n, double dt, double position, double sigma) {
62 std::vector<double> h(n);
63 double s = 0.0;
64 for (std::size_t i = 0; i < n; ++i) {
65 const double u = (double(i) * dt - position) / sigma;
66 h[i] = std::exp(-0.5 * u * u);
67 s += h[i];
68 }
69 for (double& v : h) v /= s;
70 return h;
71}

97inline BayesianTransferBasis bayesian_transfer_basis(const BayesianTransferBasisSpec& spec) {
98 const BayesianDecayAxis& ax = spec.axis;
99 const double window = double(ax.n) * ax.dt;
100 const double position = spec.response_position < 0.0 ? 0.10 * window : spec.response_position;
101 const double hi = spec.log10_tau_hi == 0.0 ? std::log10(3.0 * window) : spec.log10_tau_hi;
102 BayesianTransferBasis out;
103 out.response = bayesian_gaussian_response(ax.n, ax.dt, position, spec.response_sigma);
104 out.kernel = std::make_shared<const BayesianPeriodicKernel>(ax, bayesian_log_lifetime_grid(spec.log10_tau_lo, hi, spec.per_decade));
105 BayesianResponseOptions opt;
106 opt.remove_background = false;
107 out.basis = bayesian_response_basis(*out.kernel, out.response, 0.0, 0.0, false, opt);
108 return out;
109}

154inline std::vector<double> bayesian_transfer_rate_maps(const BayesianTransferBasis& tb,
155 const ::tttrlib::RidgeProjector& projector,
156 const std::vector<double>& rates) {
157 const BayesianResponseBasis& B = tb.basis;
158 const std::size_t n = B.n, K = B.K;
159 const std::vector<double>& tau = tb.kernel->tau();
160 const std::size_t nt = tau.size();
161 BayesianResponseOptions opt;
162 opt.remove_background = false;
163 std::vector<double> S(rates.size() * K * nt);
164 internal::bayesian_parallel_for(rates.size(), [&](std::size_t j) {
165 std::vector<double> y(n), x(K), tau_s(nt);
166 for (std::size_t c = 0; c < nt; ++c) tau_s[c] = 1.0 / (1.0 / tau[c] + rates[j]);
167 const BayesianPeriodicKernel kernel_s(tb.kernel->axis(), tau_s);
168 const BayesianResponseBasis cols = bayesian_response_basis(kernel_s, tb.response, 0.0, 0.0, false, opt);
169 for (std::size_t c = 0; c < nt; ++c) {
170 const double area = tau_s[c] / tau[c];
171 for (std::size_t i = 0; i < n; ++i) y[i] = cols.B[i * K + 1 + c] * area;
172 if (!projector.project(y.data(), x.data()))
173 throw std::runtime_error("bayesian_transfer_rate_maps: projection failed");
174 for (std::size_t k = 0; k < K; ++k) S[(j * K + k) * nt + c] = x[k];
175 }
176 });
177 return S;
178}

193class BayesianTrapezoidConvolver {
194 public:
195 BayesianTrapezoidConvolver(const double* h, std::size_t n, double dt) : n_(n), dt_(dt), h_(h, h + n) {
196 L_ = 1;
197 while (L_ < 2 * n) L_ *= 2;
198 std::vector<double> hp(L_, 0.0);
199 std::copy(h, h + n, hp.begin());
200 H_.resize(L_ / 2 + 1);
201 bayesian_rfft(hp.data(), L_, H_.data());
202 }
203 void operator()(const double* x, double* out) const {
204 std::vector<double> xp(L_, 0.0), y(L_);
205 std::copy(x, x + n_, xp.begin());
206 std::vector<std::complex<double>> X(L_ / 2 + 1);
207 bayesian_rfft(xp.data(), L_, X.data());
208 for (std::size_t k = 0; k < X.size(); ++k) X[k] *= H_[k];
209 bayesian_irfft(X.data(), L_, y.data());
210 for (std::size_t i = 0; i < n_; ++i) out[i] = std::max(dt_ * (y[i] - 0.5 * h_[i] * x[0]), 0.0);
211 }
212
213 private:
214 std::size_t n_, L_ = 1;
215 double dt_;
216 std::vector<double> h_;
217 std::vector<std::complex<double>> H_;
218};

264inline std::vector<BayesianDecayTerm> bayesian_sensitised_terms(double tau_d, double k, double tau_acc,
265 double extra_rate = 0.0, double eps = 1e-8) {
266 std::vector<BayesianDecayTerm> out;
267 const double eff = k / (1.0 / tau_d + k);
268 internal::TransferComponent<double> comp[2];
269 internal::transfer_pair_components_t<double>(tau_d, tau_acc, k, 0.0, 1.0, 0.0, 0.0, 1.0, TRANSFER_EXACT, eps, comp);
270 auto area = [](const internal::TransferComponent<double>& q, double rate) {
271 return q.kind == TRANSFER_T_EXPONENTIAL ? q.amplitude / (rate * rate) : q.amplitude / rate;
272 };
273 double S = 0.0;
274 for (const auto& q : comp) S += area(q, q.rate);
275 if (!(S > 0.0) || !(eff > 0.0)) return out;
276 for (const auto& q : comp) {
277 if (q.amplitude == 0.0) continue;
278 const double rate = q.rate + extra_rate;
279 out.push_back({1.0 / rate, q.kind, eff / S * area(q, rate)});
280 }
281 return out;
282}

329 out.sensitised_rot.assign(nh * nr * na * K * nt, 0.0);

337 internal::bayesian_parallel_for(nr, [&](std::size_t j) {
338 // every component column this distance needs, then one basis call builds them all
339 struct Term { std::size_t column; double weight; };
340 std::vector<double> life; std::vector<int> order;
341 std::vector<std::vector<Term>> terms((1 + nh) * nt * na); // [variant (0 plain, 1+r rotated)][c][l]
342 for (std::size_t c = 0; c < nt; ++c)
343 for (std::size_t l = 0; l < na; ++l)
344 for (std::size_t v = 0; v <= nh; ++v)
345 for (const auto& t : internal::bayesian_sensitised_terms(tau[c], fret_rates[j], tau_a[l],
346 v == 0 ? 0.0 : 1.0 / rho_a[v - 1], eps)) {
347 terms[(v * nt + c) * na + l].push_back({life.size(), t.weight});
348 life.push_back(t.tau); order.push_back(t.order);
349 }
350 if (life.empty()) return;
351 const BayesianPeriodicKernel kern(tb.kernel->axis(), life, order);
352 const BayesianResponseBasis cols = bayesian_response_basis(kern, tb.response, 0.0, 0.0, false, opt);
353 const std::size_t Kc = cols.K;
354 std::vector<double> y(n), x(K);
355 for (std::size_t v = 0; v <= nh; ++v)
356 for (std::size_t c = 0; c < nt; ++c)
357 for (std::size_t l = 0; l < na; ++l) {
358 std::fill(y.begin(), y.end(), 0.0);
359 for (const Term& t : terms[(v * nt + c) * na + l])
360 for (std::size_t i = 0; i < n; ++i) y[i] += t.weight * cols.B[i * Kc + 1 + t.column];
361 project(y, x.data());
362 for (std::size_t kk = 0; kk < K; ++kk) {
363 if (v == 0) out.sensitised[((j * na + l) * K + kk) * nt + c] = x[kk];
364 else out.sensitised_rot[((((v - 1) * nr + j) * na + l) * K + kk) * nt + c] = x[kk];
365 }
366 }
367 });
368 // direct excitation of the acceptor
369 std::vector<double> life, y(n), x(K);
370 for (std::size_t l = 0; l < na; ++l) {
371 life.push_back(tau_a[l]);
372 for (std::size_t r = 0; r < nh; ++r) life.push_back(1.0 / (1.0 / tau_a[l] + 1.0 / rho_a[r]));
373 }
374 const BayesianPeriodicKernel kern(tb.kernel->axis(), life);
375 const BayesianResponseBasis cols = bayesian_response_basis(kern, tb.response, 0.0, 0.0, false, opt);
376 for (std::size_t l = 0; l < na; ++l) {
377 const std::size_t base = l * (1 + nh);
378 for (std::size_t i = 0; i < n; ++i) y[i] = cols.B[i * cols.K + 1 + base];
379 project(y, x.data());
380 double area = 0.0;
381 for (std::size_t i = 0; i < n; ++i) for (std::size_t kk = 0; kk < K; ++kk) area += B.B[i * K + kk] * x[kk];
382 area = std::max(area, 1e-300);
383 for (std::size_t kk = 0; kk < K; ++kk) out.direct[l * K + kk] = x[kk] / area;
384 for (std::size_t r = 0; r < nh; ++r) {
385 const double ts = life[base + 1 + r];
386 for (std::size_t i = 0; i < n; ++i) y[i] = cols.B[i * cols.K + 2 + base + r] * (ts / tau_a[l]);
387 project(y, x.data());
388 for (std::size_t kk = 0; kk < K; ++kk) out.direct_rot[(r * na + l) * K + kk] = x[kk] / area;
389 }
390 }
391 return out;
392}

395 const ::tttrlib::RidgeProjector& projector,
396 const std::vector<double>& fret_rates,
397 const std::vector<double>& tau_a,
398 const std::vector<double>& rho_a,

400 if (how == BayesianAcceptorConstruction::exact)
401 return bayesian_transfer_acceptor_maps_exact(tb, projector, fret_rates, tau_a, rho_a);
402 const BayesianResponseBasis& B = tb.basis;
403 const std::size_t n = B.n, K = B.K;
404 const double dt = tb.kernel->axis().dt;
405 const std::vector<double>& tau = tb.kernel->tau();
406 const std::size_t nt = tau.size(), nr = fret_rates.size(), na = tau_a.size(), nh = rho_a.size();

415 if (!projector.project(y.data(), x)) throw std::runtime_error("bayesian_transfer_acceptor_maps: projection failed");
416 };
417 std::vector<std::vector<double>> kern(na, std::vector<double>(n)), fall(nh, std::vector<double>(n));
418 std::vector<internal::BayesianTrapezoidConvolver> by_kern;
419 for (std::size_t l = 0; l < na; ++l) {
420 double s = 0.0;
421 for (std::size_t i = 0; i < n; ++i) { kern[l][i] = std::exp(-double(i) * dt / tau_a[l]); s += kern[l][i]; }

429 std::vector<double> d(n), w(n), acc(n), col(n), x(K);

432 for (std::size_t i = 0; i < n; ++i) d[i] = std::exp(-double(i) * dt / tq) * bf;
433 for (std::size_t l = 0; l < na; ++l) {
434 by_kern[l](d.data(), acc.data());
435 by_response(acc.data(), col.data());
436 double tot = 0.0;
437 for (double v : col) tot += v;
438 const double scale = tot > 1e-300 ? eff / tot : 0.0;
439 for (double& v : col) v *= scale;
440 project(col, x.data());
441 for (std::size_t k = 0; k < K; ++k) out.sensitised[((j * na + l) * K + k) * nt + c] = x[k];
442 for (std::size_t r = 0; r < nh; ++r) {
443 for (std::size_t i = 0; i < n; ++i) w[i] = d[i] * fall[r][i];
444 by_kern[l](w.data(), acc.data());
445 for (std::size_t i = 0; i < n; ++i) acc[i] *= fall[r][i];
446 by_response(acc.data(), col.data());
447 for (double& v : col) v *= scale;
448 project(col, x.data());
449 for (std::size_t k = 0; k < K; ++k) out.sensitised_rot[(((r * nr + j) * na + l) * K + k) * nt + c] = x[k];
450 }
451 }
452 }
453 });
454 std::vector<double> d(n), w(n), col(n), x(K);
455 for (std::size_t l = 0; l < na; ++l) {
456 const double bf = internal::bayesian_bin_factor(tau_a[l], dt);
457 for (std::size_t i = 0; i < n; ++i) d[i] = std::exp(-double(i) * dt / tau_a[l]) * bf;
458 by_response(d.data(), col.data());
459 project(col, x.data());
460 double area = 0.0;
461 for (std::size_t i = 0; i < n; ++i) for (std::size_t k = 0; k < K; ++k) area += B.B[i * K + k] * x[k];
462 area = std::max(area, 1e-300);
463 for (std::size_t k = 0; k < K; ++k) out.direct[l * K + k] = x[k] / area;
464 for (std::size_t r = 0; r < nh; ++r) {
465 for (std::size_t i = 0; i < n; ++i) w[i] = d[i] * fall[r][i];

467 project(col, x.data());
468 for (std::size_t k = 0; k < K; ++k) out.direct_rot[(r * na + l) * K + k] = x[k] / area;
469 }
