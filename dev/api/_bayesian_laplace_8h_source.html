75inline bool bayesian_smallest_eigenpair(const double* A, std::size_t n, double* v, double* lambda_min,
76 int iterations = 60) {
77 CholeskyFactor ch;
78 if (!ch.factor(A, n)) return false;
79 std::vector<double> x(n, 0.0), y(n, 0.0);
80 for (std::size_t i = 0; i < n; ++i) x[i] = 1.0 / std::sqrt(double(n));
81 double lam = 0.0;
82 for (int it = 0; it < iterations; ++it) {
83 if (!ch.solve(x.data(), y.data())) return false; //: y = A^-1 x, so y grows along the flattest axis
84 double nrm = 0.0;
85 for (std::size_t i = 0; i < n; ++i) nrm += y[i] * y[i];
86 nrm = std::sqrt(nrm);
87 if (!(nrm > 0.0) || !std::isfinite(nrm)) return false;
88 for (std::size_t i = 0; i < n; ++i) x[i] = y[i] / nrm;
89 //: the Rayleigh quotient x' A x, which is the eigenvalue once x has settled
90 lam = 0.0;
91 for (std::size_t i = 0; i < n; ++i) {
92 double s = 0.0;
93 for (std::size_t j = 0; j < n; ++j) s += A[i * n + j] * x[j];
94 lam += x[i] * s;
95 }
96 }
97 for (std::size_t i = 0; i < n; ++i) v[i] = x[i];
98 *lambda_min = lam;
99 return true;
100}

136inline BayesianTierneyKadane bayesian_tierney_kadane(double log_ratio1, double log_ratio2, double eps) {
137 BayesianTierneyKadane r;
138 if (!std::isfinite(log_ratio1) || !std::isfinite(log_ratio2)) return r;
139 const double m1 = std::exp(log_ratio1), m2 = std::exp(log_ratio2);
140 const double var = m2 - m1 * m1;
141 if (!std::isfinite(m1) || !std::isfinite(m2)) return r;
142 r.mean = m1 - eps;
143 r.sd = std::sqrt(var > 0.0 ? var : 0.0);
144 r.ok = true;
145 return r;
146}

199 std::vector<double> weights() const {
200 std::vector<double> w(lz_.size(), 0.0);
201 if (lz_.empty()) return w;
202 double mx = -std::numeric_limits<double>::infinity();
203 for (double z : lz_) if (std::isfinite(z) && z > mx) mx = z;
204 double s = 0.0;
205 for (std::size_t i = 0; i < lz_.size(); ++i) {
206 w[i] = std::isfinite(lz_[i]) ? std::exp(lz_[i] - mx) : 0.0;
207 s += w[i];
208 }
209 if (s > 0.0) for (double& x : w) x /= s;
210 return w;
211 }

246 double cdf(double t) const {
247 const auto w = weights();
248 double F = 0.0;
249 for (std::size_t i = 0; i < w.size(); ++i) {
250 if (w[i] <= 0.0) continue;
251 const double sd = std::sqrt(v_[i] > 0.0 ? v_[i] : 0.0);
252 F += w[i] * (sd > 0.0 ? 0.5 * std::erfc(-(t - m_[i]) / (sd * 1.4142135623730951)) : (t >= m_[i] ? 1.0 : 0.0));
253 }
254 return F;
255 }

264 double quantile(double p) const {
265 const auto w = weights();
266 double lo = 0.0, hi = 0.0;
267 bool any = false;
268 for (std::size_t i = 0; i < w.size(); ++i) {
269 if (w[i] <= 0.0) continue;
270 const double sd = std::sqrt(v_[i] > 0.0 ? v_[i] : 0.0);
271 lo = any ? std::min(lo, m_[i] - 40.0 * sd) : m_[i] - 40.0 * sd;
272 hi = any ? std::max(hi, m_[i] + 40.0 * sd) : m_[i] + 40.0 * sd;
273 any = true;
274 }
275 if (!any) return std::numeric_limits<double>::quiet_NaN();
276 if (!(p > 0.0)) return lo;
277 if (!(p < 1.0)) return hi;
278 for (int it = 0; it < 200; ++it) {
279 const double mid = 0.5 * (lo + hi);
280 (cdf(mid) < p ? lo : hi) = mid;
281 }
282 return 0.5 * (lo + hi);
283 }
