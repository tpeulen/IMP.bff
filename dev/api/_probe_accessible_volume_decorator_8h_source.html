68 /* Ref-counted, not a bare pointer. AV is an IMP *decorator* -- a value
69 handle constructed, copied and discarded freely -- and it was holding a
70 raw `new PathMap` with no destructor on this class at all. Every fresh
71 handle over the same particle built and then abandoned an entire path
72 map: measured at ~1 MB a time, 21 MB for twenty handles. A destructor
73 would have been worse, not better, because two copies of a decorator
74 would then free the same map twice. PathMap derives from IMP::Object and
75 is therefore already ref-counted, so IMP::Pointer gives copies a shared
76 map and frees it when the last handle goes. */

325 void set_chain_weighting(bool on) {
326 if (get_model()->get_has_attribute(get_chain_weighting_key(),
327 get_particle_index())) {
328 get_model()->set_attribute(get_chain_weighting_key(),
329 get_particle_index(), on ? 1 : 0);
330 } else {
331 get_model()->add_attribute(get_chain_weighting_key(),
332 get_particle_index(), on ? 1 : 0);
333 }
334 if (!on) {
335 if (av_map_) av_map_->set_linker_weighting(LinkerWeighting());
336 return;
337 }
338 const LinkerWeighting w = linker_weighting(get_linker_length());
339 const double reach = w.get_supported_fraction(get_linker_length());
340 // A table column fitted for a much longer tether puts none of its mass
341 // inside a dye linker's reach. Weighting by it then does not correct
342 // the volume, it selects the volume's outer shell -- so say so rather
343 // than let a 13 A shift in the mean position pass for a refinement.
344 if (reach < 0.01) {
345 IMP_WARN("chain_weighting: the tabulated weighting for a linker of "
346 << get_linker_length() << " A carries only "
347 << reach * 100.0
348 << "% of its weight within that reach, so it selects the "
349 "volume's outer shell rather than reweighting it. See "
350 "okf/validation/chain_weighting.md."
351 << std::endl);
352 }
353 if (av_map_) av_map_->set_linker_weighting(w);
354 }

803template<typename T>
804T inline fret_efficiency(T distance, double forster_radius){
805 // (r/R0)^6 by three multiplications; std::pow(x, 6.0) costs ~20x more
806 double x = distance / forster_radius;
807 double x2 = x * x;
808 double rda_r0_6 = x2 * x2 * x2;
809 return 1. / (1. + rda_r0_6);
810}

819template<typename T>
820T inline distance_fret(double fret_efficiency, double forster_radius){
821 return forster_radius * std::pow(1. / fret_efficiency - 1.0, 1. / 6.);
822}
