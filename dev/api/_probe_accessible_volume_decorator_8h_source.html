68 /* Ref-counted, not a bare pointer. AV is an IMP *decorator* -- a value
69 handle constructed, copied and discarded freely -- and it was holding a
70 raw `new PathMap` with no destructor on this class at all. Every fresh
71 handle over the same particle built and then abandoned an entire path
72 map: measured at ~1 MB a time, 21 MB for twenty handles. A destructor
73 would have been worse, not better, because two copies of a decorator
74 would then free the same map twice. PathMap derives from IMP::Object and
75 is therefore already ref-counted, so IMP::Pointer gives copies a shared
76 map and frees it when the last handle goes. */

326 void set_chain_weighting(bool on) {
327 if (get_model()->get_has_attribute(get_chain_weighting_key(),
328 get_particle_index())) {
329 get_model()->set_attribute(get_chain_weighting_key(),
330 get_particle_index(), on ? 1 : 0);
331 } else {
332 get_model()->add_attribute(get_chain_weighting_key(),
333 get_particle_index(), on ? 1 : 0);
334 }
335 if (!on) {
336 if (av_map_) av_map_->set_linker_weighting(LinkerWeighting());
337 return;
338 }
339 const LinkerWeighting w = linker_weighting(get_linker_length());
340 const double reach = w.get_supported_fraction(get_linker_length());
341 // A table column fitted for a much longer tether puts none of its mass
342 // inside a dye linker's reach. Weighting by it then does not correct
343 // the volume, it selects the volume's outer shell -- so say so rather
344 // than let a 13 A shift in the mean position pass for a refinement.
345 if (reach < 0.01) {
346 IMP_WARN("chain_weighting: the tabulated weighting for a linker of "
347 << get_linker_length() << " A carries only "
348 << reach * 100.0
349 << "% of its weight within that reach, so it selects the "
350 "volume's outer shell rather than reweighting it. See "
351 "okf/validation/chain_weighting.md."
352 << std::endl);
353 }
354 if (av_map_) av_map_->set_linker_weighting(w);
355 }

841template<typename T>
842T inline fret_efficiency(T distance, double forster_radius){
843 // (r/R0)^6 by three multiplications; std::pow(x, 6.0) costs ~20x more
844 double x = distance / forster_radius;
845 double x2 = x * x;
846 double rda_r0_6 = x2 * x2 * x2;
847 return 1. / (1. + rda_r0_6);
848}

857template<typename T>
858T inline distance_fret(double fret_efficiency, double forster_radius){
859 return forster_radius * std::pow(1. / fret_efficiency - 1.0, 1. / 6.);
860}
