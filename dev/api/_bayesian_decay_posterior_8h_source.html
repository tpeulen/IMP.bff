74inline BayesianDecayPrior bayesian_decay_log_prior(const BayesianDecayExperiment& f, const std::vector<double>& th, bool derivs = true) {
75 const std::size_t dim = f.dim;
76 const double l2pi = std::log(2.0 * 3.14159265358979323846);
77 BayesianDecayPrior P;
78 if (derivs) { P.g.assign(dim, 0.0); P.H.assign(dim * dim, 0.0); }
79 for (const BayesianDecayVariable& v : f.variables) {
80 const std::string& fam = v.family, & tr = v.transform, & nm = v.name;
81 const std::size_t o = v.offset, sz = v.size;
82 if (fam == "pspline") continue; // c: the P-spline term below
83 if (fam == "spectrum_smoothness") {
84 const BayesianDecayArray& Pm = f["spec_smooth_P"];
85 const double logdet = f.spec_smooth_logdet;
86 double q = 0.0;
87 for (std::size_t i = 0; i < sz; ++i) {
88 double r = 0.0; for (std::size_t j = 0; j < sz; ++j) r += Pm.d[i * sz + j] * th[o + j];
89 q += th[o + i] * r;
90 if (derivs) { P.g[o + i] -= r; for (std::size_t j = 0; j < sz; ++j) P.H[(o + i) * dim + o + j] -= Pm.d[i * sz + j]; }
91 }
92 P.lp += 0.5 * logdet - 0.5 * q - 0.5 * double(sz) * l2pi;
93 continue;
94 }
95 for (std::size_t i = 0; i < sz; ++i) {
96 const double z = th[o + i], a = v.a[i], b = v.b[i];
97 double lp = 0.0, g = 0.0, h = 0.0;
98 if (fam == "gaussian_on_z" || (fam == "gaussian" && tr == "identity")) {
99 lp = -0.5 * ((z - a) / b) * ((z - a) / b) - std::log(b) - 0.5 * l2pi; g = -(z - a) / (b * b); h = -1.0 / (b * b);
100 } else if (fam == "lognormal" && tr == "log") {
101 //: log x = z: the prior's -log x and the transform's +z cancel
102 const double u = (z - std::log(a)) / b;
103 lp = -0.5 * u * u - std::log(b) - 0.5 * l2pi; g = -u / b; h = -1.0 / (b * b);
104 } else if (fam == "uniform" && tr == "logit") {
105 const double lo = v.lo, hi = v.hi;
106 //: -log(hi - lo) from the prior, log(hi - lo) + logsigmoid(z) + logsigmoid(-z) from the transform
107 lp = -bayesian_soft_positive(-z, 1.0, BAYESIAN_TORCH_SOFTPLUS_THRESHOLD) - bayesian_soft_positive(z, 1.0, BAYESIAN_TORCH_SOFTPLUS_THRESHOLD);
108 const double s = 1.0 / (1.0 + std::exp(-z));
109 g = 1.0 - 2.0 * s; h = -2.0 * s * (1.0 - s);
110 (void)lo; (void)hi;
111 } else {
112 throw std::runtime_error("bayesian_decay_log_prior: family " + fam + " on transform " + tr + " (" + nm + ") is not implemented");
113 }
114 P.lp += lp;
115 if (derivs) { P.g[o + i] += g; P.H[(o + i) * dim + o + i] += h; }
116 }
117 }
118 // the P-spline on c = Q z
119 const BayesianDecayPSpline& ps = f.pspline;
120 if (ps.order != 2 || ps.family != "gaussian" || ps.space != "log")
121 throw std::runtime_error("bayesian_decay_log_prior: only the order-2 Gaussian P-spline in log space is implemented");
122 const BayesianDecayVariableInfo vc = bayesian_decay_variable_info(f, "c");
123 const BayesianDecayArray& Q = f["Q_c"];
124 const std::size_t n = Q.shape[0], nz = Q.shape[1];
125 //: lambda: held (fixed_values) or free (a variable `log10_lam`, logit on [lo, hi] -- its own
126 //: prior handled with the other variables above); PRD-143 A4.6
127 const BayesianDecayVariableInfo vl = bayesian_decay_variable_info(f, "log10_lam");
128 const auto fixed_lam = f.fixed_values.find("log10_lam");
129 if (vl.present == (fixed_lam != f.fixed_values.end()))
130 throw std::runtime_error("bayesian_decay_log_prior: log10_lam must be either held or free, exactly one");
131 double x_lam = 0.0, dL_dz = 0.0, d2L_dz2 = 0.0; // L = ln lambda as a function of its z
132 if (vl.present) {
133 const BayesianDecayVariable* var = f.variable("log10_lam");
134 if (var->transform != "logit") throw std::runtime_error("bayesian_decay_log_prior: a free log10_lam needs the logit transform");
135 const double z = th[vl.off], sg = 1.0 / (1.0 + std::exp(-z)), w = var->hi - var->lo, ln10 = std::log(10.0);
136 x_lam = var->lo + w * sg;
137 dL_dz = ln10 * w * sg * (1.0 - sg);
138 d2L_dz2 = ln10 * w * sg * (1.0 - sg) * (1.0 - 2.0 * sg);
139 } else {
140 x_lam = fixed_lam->second[0];
141 }
142 const double lam = std::pow(10.0, x_lam);
143 const double tilt_sd = ps.tilt_sd, rank = ps.rank;
144 std::vector<double> c(n, 0.0);
145 for (std::size_t i = 0; i < n; ++i) for (std::size_t j = 0; j < nz; ++j) c[i] += Q.d[i * nz + j] * th[vc.off + j];
146 //: BayesianPSplinePrior (Eilers & Marx 1996): second differences with precision lambda,
147 //: a weak Gaussian on the tilt, (rank/2) log lambda; its gradient and Hessian in c,
148 //: carried to z through Q
149 const BayesianPSplinePrior<double> prior(n, ps.order, tilt_sd, ps.quad_sd);
150 if (std::size_t(rank) != std::size_t(prior.rank()))
151 throw std::runtime_error("bayesian_decay_log_prior: the manifest's P-spline rank disagrees with the prior's");
152 P.lp += prior.log_prob(c.data(), lam);
153 if (derivs) {
154 std::vector<double> gc(n), Hc(n * n);
155 prior.gradient(c.data(), lam, gc.data());
156 prior.hessian(c.data(), lam, Hc.data());
157 for (std::size_t j = 0; j < nz; ++j) { double s = 0.0; for (std::size_t i = 0; i < n; ++i) s += Q.d[i * nz + j] * gc[i]; P.g[vc.off + j] += s; }
158 std::vector<double> HQ(n * nz, 0.0);
159 for (std::size_t i = 0; i < n; ++i) for (std::size_t j = 0; j < nz; ++j) { double s = 0.0; for (std::size_t k = 0; k < n; ++k) s += Hc[i * n + k] * Q.d[k * nz + j]; HQ[i * nz + j] = s; }
160 for (std::size_t a = 0; a < nz; ++a) for (std::size_t b = 0; b < nz; ++b) {
161 double s = 0.0; for (std::size_t i = 0; i < n; ++i) s += Q.d[i * nz + a] * HQ[i * nz + b];
162 P.H[(vc.off + a) * dim + vc.off + b] += s;
163 }
164 }
165 if (vl.present && derivs) {
166 //: in L = ln lambda the prior is (rank/2) L - (lambda/2) q(c) - tilt(c), q = |D c|^2:
167 //: f' = rank/2 - lambda q / 2, f'' = -lambda q / 2, d f'/dc = -lambda D'D c
168 const std::vector<double>& D = prior.difference_matrix();
169 const std::size_t nr = n - std::size_t(ps.order);
170 std::vector<double> d(nr, 0.0), DtDc(n, 0.0);
171 double q = 0.0;
172 for (std::size_t k = 0; k < nr; ++k) { for (std::size_t i = 0; i < n; ++i) d[k] += D[k * n + i] * c[i]; q += d[k] * d[k]; }
173 for (std::size_t k = 0; k < nr; ++k) for (std::size_t i = 0; i < n; ++i) DtDc[i] += D[k * n + i] * d[k];
174 const double f1 = 0.5 * rank - 0.5 * lam * q, f2 = -0.5 * lam * q;
175 const std::size_t zl = vl.off;
176 P.g[zl] += f1 * dL_dz;
177 P.H[zl * dim + zl] += f2 * dL_dz * dL_dz + f1 * d2L_dz2;
178 for (std::size_t j = 0; j < nz; ++j) {
179 double s = 0.0; for (std::size_t i = 0; i < n; ++i) s += Q.d[i * nz + j] * DtDc[i];
180 const double h = -lam * s * dL_dz;
181 P.H[(vc.off + j) * dim + zl] += h;
182 P.H[zl * dim + vc.off + j] += h;
183 }
184 }
185 return P;
186}

188inline double bayesian_decay_log_likelihood(const BayesianDecayExperiment& f, const std::vector<double>& lam) {
189 const BayesianDecayArray& y = f["y"], & mask = f["mask"];
190 double ll = 0.0;
191 for (std::size_t i = 0; i < lam.size(); ++i) {
192 if (mask.d[i] == 0.0) continue;
193 const double l = std::max(lam[i], 1e-12);
194 ll += mask.d[i] * (y.d[i] * std::log(l) - l);
195 }
196 return ll;
197}

210inline BayesianDecayPoint bayesian_decay_evaluate(const BayesianDecayExperiment& f, const BayesianDecayTensors& e, const std::vector<double>& th) {
211 const std::size_t dim = f.dim;
212 BayesianDecayPoint p; p.theta = th;
213 const BayesianDecayValues v = bayesian_decay_unpack(f, th);
214 const std::vector<double> a2 = bayesian_decay_amplitudes(f, e, v);
215 const std::vector<double> Ja = bayesian_decay_amplitude_jacobian(f, e, v);
216 std::vector<std::vector<std::size_t>> cols;
217 p.lam = bayesian_decay_expected_counts(f, e, v, &p.J, &Ja, &a2, &cols);
218 const BayesianDecayPrior P = bayesian_decay_log_prior(f, th, true);
219 p.logprior = P.lp;
220 p.logpost = bayesian_decay_log_likelihood(f, p.lam) + P.lp;
221 //: the likelihood's gradient and Fisher matrix through tttrlib::poisson_score_blocks:
222 //: each histogram on its live columns, the histograms on the thread pool
223 const BayesianDecayArray& y = f["y"], & mask = f["mask"];
224 p.grad.assign(dim, 0.0);
225 p.A.assign(dim * dim, 0.0);
226 ::tttrlib::poisson_score_blocks(y.d.data(), p.lam.data(), p.J.data(), mask.d.data(), p.lam.size(), dim,
227 ::tttrlib::POISSON_INFORMATION_EXPECTED, cols, f["y"].shape[1], p.grad.data(), p.A.data(),
228 [](std::size_t n, const std::function<void(std::size_t)>& body) { internal::bayesian_parallel_for(n, body); });
229 for (std::size_t a = 0; a < dim; ++a) p.grad[a] += P.g[a];
230 for (std::size_t k = 0; k < dim * dim; ++k) p.A[k] -= P.H[k];
231 return p;
232}

237inline double bayesian_decay_log_posterior_and_gradient(const BayesianDecayExperiment& f, const BayesianDecayTensors& e,
238 const std::vector<double>& th, std::vector<double>& grad) {
239 const std::size_t dim = f.dim;
240 const BayesianDecayValues v = bayesian_decay_unpack(f, th);
241 const std::vector<double> a2 = bayesian_decay_amplitudes(f, e, v);
242 const std::vector<double> Ja = bayesian_decay_amplitude_jacobian(f, e, v);
243 std::vector<double> J;
244 const std::vector<double> lam = bayesian_decay_expected_counts(f, e, v, &J, &Ja, &a2);
245 const BayesianDecayPrior P = bayesian_decay_log_prior(f, th, true);
246 const BayesianDecayArray& y = f["y"], & mask = f["mask"];
247 grad.assign(dim, 0.0);
248 double ll = 0.0;
249 for (std::size_t b = 0; b < lam.size(); ++b) {
250 const double w = mask.d[b];
251 if (w == 0.0) continue;
252 const double m = std::max(lam[b], 1e-12);
253 ll += w * (y.d[b] * std::log(m) - m);
254 const double u = w * (y.d[b] / m - 1.0);
255 if (u == 0.0) continue;
256 const double* row = &J[b * dim];
257 for (std::size_t c = 0; c < dim; ++c) grad[c] += u * row[c];
258 }
259 for (std::size_t c = 0; c < dim; ++c) grad[c] += P.g[c];
260 return ll + P.lp;
261}

276inline std::vector<double> bayesian_decay_covariance(const BayesianDecayPoint& p, std::size_t dim) {
277 CholeskyFactor ch;
278 std::vector<double> S(dim * dim, 0.0), e(dim, 0.0), col(dim);
279 if (!ch.factor(p.A.data(), dim)) return {};
280 for (std::size_t j = 0; j < dim; ++j) {
281 std::fill(e.begin(), e.end(), 0.0); e[j] = 1.0;
282 ch.solve(e.data(), col.data());
283 for (std::size_t i = 0; i < dim; ++i) S[i * dim + j] = col[i];
284 }
285 return S;
286}

290inline void bayesian_decay_distribution_with_sd(const BayesianDecayExperiment& f, const BayesianDecayTensors& e, const std::vector<double>& th, const std::vector<double>& Sig,
291 std::vector<double>& p, std::vector<double>& sd, double* mean_rel = nullptr, double* mean_rel_sd = nullptr) {
292 const std::size_t dim = f.dim;
293 p = bayesian_decay_distribution(e, bayesian_decay_unpack(f, th));
294 const BayesianDecayVariableInfo vc = bayesian_decay_variable_info(f, "c");
295 const BayesianDecayArray& Q = f["Q_c"];
296 const std::size_t nR = p.size(), nc = e.spl->shape[1], nz = Q.shape[1];
297 std::vector<double> BQ(nR * nz, 0.0), pBQ(nz, 0.0), Jp(nR * nz);
298 for (std::size_t j = 0; j < nR; ++j) for (std::size_t a = 0; a < nc; ++a) {
299 const double s_ = e.spl->d[j * nc + a];
300 if (s_ != 0.0) for (std::size_t b = 0; b < nz; ++b) BQ[j * nz + b] += s_ * Q.d[a * nz + b];
301 }
302 for (std::size_t j = 0; j < nR; ++j) for (std::size_t b = 0; b < nz; ++b) pBQ[b] += p[j] * BQ[j * nz + b];
303 for (std::size_t j = 0; j < nR; ++j) for (std::size_t b = 0; b < nz; ++b) Jp[j * nz + b] = p[j] * (BQ[j * nz + b] - pBQ[b]);
304 //: the delta method through bayesian_delta_variance, on the c block of Sigma
305 std::vector<double> Scc(nz * nz);
306 for (std::size_t a = 0; a < nz; ++a) for (std::size_t b = 0; b < nz; ++b) Scc[a * nz + b] = Sig[(vc.off + a) * dim + vc.off + b];
307 sd.assign(nR, 0.0);
308 bayesian_delta_variance(Jp.data(), Scc.data(), nR, nz, sd.data());
309 for (double& t : sd) t = std::sqrt(t);
310 if (mean_rel) {
311 //: the mean of R/R0 and its delta-method sd: d mean / d z = rel' J_p
312 const std::vector<double>& rel = f["rel"].d;
313 std::vector<double> jm(nz, 0.0);
314 double m = 0.0;
315 for (std::size_t j = 0; j < nR; ++j) { m += p[j] * rel[j]; for (std::size_t b = 0; b < nz; ++b) jm[b] += rel[j] * Jp[j * nz + b]; }
316 double v = 0.0;
317 bayesian_delta_variance(jm.data(), Scc.data(), 1, nz, &v);
318 *mean_rel = m;
319 if (mean_rel_sd) *mean_rel_sd = std::sqrt(v);
320 }

344inline BayesianDecayTilt bayesian_decay_mean_rel_tilt(const BayesianDecayExperiment& f, const BayesianDecayTensors& e,
345 const std::vector<double>& th, double power, double eps,
346 bool want_hessian = true) {
347 const std::size_t dim = f.dim;
348 BayesianDecayTilt T;
349 T.grad.assign(dim, 0.0);
350 if (want_hessian) T.hess.assign(dim * dim, 0.0);
351 const std::vector<double> p = bayesian_decay_distribution(e, bayesian_decay_unpack(f, th));
352 const std::vector<double>& rel = f["rel"].d;
353 const BayesianDecayVariableInfo vc = bayesian_decay_variable_info(f, "c");
354 const BayesianDecayArray& Q = f["Q_c"];
355 const std::size_t nR = p.size(), nc = e.spl->shape[1], nz = Q.shape[1];
356 std::vector<double> M(nR * nz, 0.0); // spl @ Q, so u = M z
357 for (std::size_t j = 0; j < nR; ++j) for (std::size_t a = 0; a < nc; ++a) {
358 const double s_ = e.spl->d[j * nc + a];
359 if (s_ != 0.0) for (std::size_t b = 0; b < nz; ++b) M[j * nz + b] += s_ * Q.d[a * nz + b];
360 }
361 double fv = 0.0;
362 for (std::size_t j = 0; j < nR; ++j) fv += p[j] * rel[j];
363 T.f = fv;
364 std::vector<double> gj(nR), G(nz, 0.0), P(nz, 0.0);
365 for (std::size_t j = 0; j < nR; ++j) gj[j] = p[j] * (rel[j] - fv);
366 for (std::size_t j = 0; j < nR; ++j) for (std::size_t b = 0; b < nz; ++b) {
367 G[b] += gj[j] * M[j * nz + b];
368 P[b] += p[j] * M[j * nz + b];
369 }
370 const double d = fv + eps, k1 = power / d;
371 T.value = power * std::log(d);
372 for (std::size_t b = 0; b < nz; ++b) T.grad[vc.off + b] = k1 * G[b];
373 if (want_hessian) {
374 std::vector<double> W(nz * nz, 0.0); // M' diag(g) M
375 for (std::size_t j = 0; j < nR; ++j) {
376 if (gj[j] == 0.0) continue;
377 for (std::size_t a = 0; a < nz; ++a) {
378 const double t = gj[j] * M[j * nz + a];
379 for (std::size_t b = 0; b < nz; ++b) W[a * nz + b] += t * M[j * nz + b];
380 }
381 }
382 const double k2 = power / (d * d);
383 for (std::size_t a = 0; a < nz; ++a) for (std::size_t b = 0; b < nz; ++b) {
384 const double d2f = W[a * nz + b] - G[a] * P[b] - P[a] * G[b];
385 T.hess[(vc.off + a) * dim + vc.off + b] = k1 * d2f - k2 * G[a] * G[b];
386 }
387 }
388 return T;
389}

394struct BayesianDecayFit {
395 std::vector<double> theta;
396 BayesianDecayPoint pt;
397 double evidence = -std::numeric_limits<double>::infinity(), dec = std::nan(""), t_eval = 0.0, t_search = 0.0, t_fit = 0.0;
398 bool converged = false, chol = false;
399 int iterations = 0, n_obj = 0, n_fallback = 0;

402inline BayesianDecayFit bayesian_decay_fit_node(const BayesianDecayExperiment& f, const BayesianDecayTensors& env, std::vector<double> th, double line_search_below = 1.0,
403 int max_iter = 300, double tol = 1e-6, bool verbose = false) {
404 using clk_ = std::chrono::steady_clock;
405 auto secs_ = [](clk_::time_point a) { return std::chrono::duration<double>(clk_::now() - a).count(); };
406 const auto t0 = clk_::now();
407 const std::size_t dim = f.dim;
408 BayesianDecayFit R;
409 BayesianDecayPoint pt = bayesian_decay_evaluate(f, env, th);
410 int it = 0;
411 auto objective = [&](const double* t) { ++R.n_obj; return -bayesian_decay_log_posterior(f, env, std::vector<double>(t, t + dim)); };
412 DampedNewton<decltype(objective)> stepper;
413 stepper.line_search_below = line_search_below;
414 for (; it < max_iter; ++it) {
415 const auto ts = clk_::now();
416 const OptimizationStepResult r = stepper.step(pt.A.data(), pt.grad.data(), dim, -pt.logpost, th.data(), objective);
417 R.t_search += secs_(ts);
418 if (!r.accepted) { R.converged = std::isfinite(R.dec) && R.dec < 1e-3; break; }
419 R.n_fallback += r.gradient_fallback ? 1 : 0;
420 R.dec = r.decrement;
421 const auto te = clk_::now();
422 pt = bayesian_decay_evaluate(f, env, th);
423 R.t_eval += secs_(te);
424 if (verbose) std::printf(" step %3d: log post %.4f, decrement %.3e, mu %.1e, alpha %.3g%s\n", it, pt.logpost, R.dec, r.mu, r.alpha, r.gradient_fallback ? " (gradient)" : "");
425 if (R.dec < tol) { R.converged = true; ++it; break; }
426 }
427 if (!R.converged && std::isfinite(R.dec) && R.dec < 1e-3) R.converged = true;
428 R.iterations = it;
429 R.evidence = bayesian_decay_laplace_evidence(pt, dim, &R.chol);
430 R.theta = std::move(th); R.pt = std::move(pt);
431 R.t_fit = secs_(t0);
432 return R;
433}

450inline std::vector<double> bayesian_decay_hessian(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
451 const std::vector<double>& theta, double rel = 1e-5) {
452 const std::size_t dim = f.dim;
453 std::vector<double> H(dim * dim, 0.0);
454 for (std::size_t j = 0; j < dim; ++j) {
455 const double h = rel * std::max(1.0, std::fabs(theta[j]));
456 std::vector<double> tp = theta, tm = theta;
457 tp[j] += h; tm[j] -= h;
458 const BayesianDecayPoint gp = bayesian_decay_evaluate(f, env, tp), gm = bayesian_decay_evaluate(f, env, tm);
459 for (std::size_t i = 0; i < dim; ++i) H[i * dim + j] = -(gp.grad[i] - gm.grad[i]) / (2.0 * h); // grad is of +log posterior
460 }
461 for (std::size_t i = 0; i < dim; ++i)
462 for (std::size_t j = i + 1; j < dim; ++j) {
463 const double v = 0.5 * (H[i * dim + j] + H[j * dim + i]);
464 H[i * dim + j] = H[j * dim + i] = v;
465 }
466 return H;
467}

478inline BayesianDecayFit bayesian_decay_polish_exact(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
479 BayesianDecayFit fit, int max_iter = 20, double tol = 1e-8) {
480 const std::size_t dim = f.dim;
481 std::vector<double> th = fit.theta;
482 auto objective = [&](const double* t) { return -bayesian_decay_log_posterior(f, env, std::vector<double>(t, t + dim)); };
483 DampedNewton<decltype(objective)> stepper;
484 BayesianDecayPoint pt = fit.pt;
485 int it = 0;
486 for (; it < max_iter; ++it) {
487 const std::vector<double> H = bayesian_decay_hessian(f, env, th);
488 const OptimizationStepResult r = stepper.step(H.data(), pt.grad.data(), dim, -pt.logpost, th.data(), objective);
489 if (!r.accepted) break;
490 pt = bayesian_decay_evaluate(f, env, th);
491 fit.dec = r.decrement;
492 if (r.decrement < tol) { ++it; break; }
493 }
494 fit.theta = th; fit.pt = std::move(pt); fit.iterations += it;
495 return fit;
496}

500inline double bayesian_decay_laplace_evidence_exact(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
501 const BayesianDecayPoint& mode, bool* ok = nullptr) {
502 BayesianDecayPoint q = mode;
503 q.A = bayesian_decay_hessian(f, env, mode.theta);
504 return bayesian_decay_laplace_evidence(q, f.dim, ok);
505}

515inline BayesianDecayFit bayesian_decay_fit_node_tilted(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
516 std::vector<double> th, double power, double eps,
517 int max_iter = 100, double tol = 1e-8) {
518 const std::size_t dim = f.dim;
519 BayesianDecayFit R;
520 auto tilted_point = [&](const std::vector<double>& t) {
521 BayesianDecayPoint pt = bayesian_decay_evaluate(f, env, t);
522 const BayesianDecayTilt T = bayesian_decay_mean_rel_tilt(f, env, t, power, eps);
523 pt.logpost += T.value;
524 for (std::size_t i = 0; i < dim; ++i) pt.grad[i] += T.grad[i];
525 for (std::size_t i = 0; i < dim * dim; ++i) pt.A[i] -= T.hess[i]; //: A is the curvature of MINUS the log posterior
526 return pt;
527 };
528 auto objective = [&](const double* t) {
529 const std::vector<double> v(t, t + dim);
530 ++R.n_obj;
531 return -(bayesian_decay_log_posterior(f, env, v) + bayesian_decay_mean_rel_tilt(f, env, v, power, eps, false).value);
532 };
533 BayesianDecayPoint pt = tilted_point(th);
534 DampedNewton<decltype(objective)> stepper;
535 int it = 0;
536 for (; it < max_iter; ++it) {
537 const OptimizationStepResult r = stepper.step(pt.A.data(), pt.grad.data(), dim, -pt.logpost, th.data(), objective);
538 if (!r.accepted) { R.converged = std::isfinite(R.dec) && R.dec < 1e-3; break; }
539 R.dec = r.decrement;
540 R.n_fallback += r.gradient_fallback ? 1 : 0;
541 pt = tilted_point(th);
542 if (R.dec < tol) { R.converged = true; ++it; break; }
543 }
544 if (!R.converged && std::isfinite(R.dec) && R.dec < 1e-3) R.converged = true;
545 R.iterations = it;
546 R.theta = th;
547 R.pt = bayesian_decay_evaluate(f, env, th);
548 return R;
549}

552struct BayesianDecayTkSummary {
553 double mean = std::nan(""), sd = std::nan("");
554 bool ok = false;
555 int iterations = 0;
556 double seconds = 0.0;
561 double shift[2] = {std::nan(""), std::nan("")};
562 double dlogdet[2] = {std::nan(""), std::nan("")};
572 double flat_curvature = std::nan("");
573 double tilt_on_flat[2] = {std::nan(""), std::nan("")};
574 double flat_ratio[2] = {std::nan(""), std::nan("")};
579 double flat_share[2] = {std::nan(""), std::nan("")};
586 double flat_prior_share = std::nan("");
587};
588
602inline BayesianDecayTkSummary bayesian_decay_tk_mean_rel(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
603 const BayesianDecayFit& node, double eps = 1e-6, int max_iter = 100,
604 bool exact_curvature = false) {
605 using clk_ = std::chrono::steady_clock;
606 const auto t0 = clk_::now();
607 BayesianDecayTkSummary out;
608 const std::size_t dim = f.dim;
609 //: the base curvature both integrals share: Fisher's matrix (the prototype's rule) or the exact Hessian
610 const std::vector<double> A0 = exact_curvature ? bayesian_decay_hessian(f, env, node.theta) : node.pt.A;
611 double logdet0 = 0.0;
612 if (!bayesian_log_det_spd(A0.data(), dim, &logdet0)) return out;
613 //: the node's flattest direction, against which the tilt's curvature is measured below
614 std::vector<double> vflat(dim, 0.0);
615 double lam_min = std::nan("");
616 const bool have_flat = bayesian_smallest_eigenpair(A0.data(), dim, vflat.data(), &lam_min);
617 out.flat_curvature = have_flat ? lam_min : std::nan("");
618 if (have_flat && lam_min > 0.0) {
619 //: how much of that direction's curvature the prior alone supplies
620 const BayesianDecayPrior pr = bayesian_decay_log_prior(f, node.theta, true);
621 double q = 0.0;
622 for (std::size_t i = 0; i < dim; ++i) {
623 double si = 0.0;
624 for (std::size_t j = 0; j < dim; ++j) si -= pr.H[i * dim + j] * vflat[j];
625 q += vflat[i] * si;
626 }
627 out.flat_prior_share = q / lam_min;
628 }
629 const double L0 = node.pt.logpost;
630 double lr[2] = {std::nan(""), std::nan("")};
631 for (int k = 0; k < 2; ++k) {
632 const double power = double(k + 1);
633 const BayesianDecayFit t = bayesian_decay_fit_node_tilted(f, env, node.theta, power, eps, max_iter);
634 out.iterations += t.iterations;
635 if (!t.converged || !std::isfinite(t.pt.logpost)) return out;
636 const BayesianDecayTilt T = bayesian_decay_mean_rel_tilt(f, env, t.theta, power, eps);
637 std::vector<double> H = A0; //: the NODE's matrix ...
638 for (std::size_t i = 0; i < dim * dim; ++i) H[i] -= T.hess[i]; //: ... plus the tilt's exact curvature at the tilted mode
639 double logdet1 = 0.0;
640 if (!bayesian_log_det_spd(H.data(), dim, &logdet1)) return out;
641 lr[k] = (t.pt.logpost + T.value) - L0 - 0.5 * (logdet1 - logdet0);
642 out.dlogdet[k] = logdet1 - logdet0;
643 double d2 = 0.0;
644 for (std::size_t i = 0; i < dim; ++i) {
645 const double di = t.theta[i] - node.theta[i];
646 if (di == 0.0) continue;
647 for (std::size_t j = 0; j < dim; ++j) d2 += di * A0[i * dim + j] * (t.theta[j] - node.theta[j]);
648 }
649 out.shift[k] = std::sqrt(d2 > 0.0 ? d2 : 0.0);
650 if (have_flat) {
651 //: v' (-d2 tilt) v, the tilt's curvature along the direction the data leave flat
652 double q = 0.0;
653 for (std::size_t i = 0; i < dim; ++i) {
654 double si = 0.0;
655 for (std::size_t j = 0; j < dim; ++j) si -= T.hess[i * dim + j] * vflat[j];
656 q += vflat[i] * si;
657 }
658 out.tilt_on_flat[k] = q;
659 out.flat_ratio[k] = lam_min > 0.0 ? std::fabs(q) / lam_min : std::numeric_limits<double>::infinity();
660 if (lam_min > 0.0 && std::fabs(out.dlogdet[k]) > 0.0)
661 out.flat_share[k] = std::log1p(q / lam_min) / out.dlogdet[k];
662 }
663 }
664 const BayesianTierneyKadane r = bayesian_tierney_kadane(lr[0], lr[1], eps);
665 out.mean = r.mean; out.sd = r.sd; out.ok = r.ok;
666 out.seconds = std::chrono::duration<double>(clk_::now() - t0).count();
667 return out;
668}

699 std::vector<BayesianDecayFit> fits;
700 std::size_t best = 0;

711inline BayesianDecayMultiFit bayesian_decay_fit_best_of_starts(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
712 const std::vector<BayesianDecayStart>& starts, int max_iter = 1000,
713 std::size_t n_concurrent = 4) {
714 const std::size_t dim = f.dim;
715 BayesianDecayMultiFit out;
716 out.fits.resize(starts.size());
717 std::atomic<std::size_t> next{0};
718 auto work = [&] {
719 internal::bayesian_serial_here() = true;
720 for (;;) {
721 const std::size_t i = next.fetch_add(1);
722 if (i >= starts.size()) return;
723 std::vector<double> th = starts[i].theta;
724 if (starts[i].adam_steps > 0) {
725 ::IMP::bff::internal::AdamState st;
726 st.reset(dim);
727 std::vector<double> neg(dim);
728 std::vector<double> g_;
729 for (int k = 0; k < starts[i].adam_steps; ++k) {
730 //: the GRADIENT path, not `evaluate`: Adam uses only the gradient, and evaluate also
731 //: builds the (n_data x dim) Jacobian and the dim x dim normal equations it throws away.
732 //: Measured on CBM56 (6 histograms, dim 136): 14.5 ms a step against 3.6 ms.
733 bayesian_decay_log_posterior_and_gradient(f, env, th, g_);
734 for (std::size_t a = 0; a < dim; ++a) neg[a] = -g_[a];
735 ::IMP::bff::internal::adam_update(th.data(), neg.data(), dim, st, starts[i].adam_lr);
736 }
737 }
738 out.fits[i] = bayesian_decay_fit_node(f, env, th, starts[i].line_search_below, max_iter);
739 }
740 };
741 std::vector<std::thread> pool;
742 for (std::size_t w = 0; w < std::max<std::size_t>(1, std::min(n_concurrent, starts.size())); ++w) pool.emplace_back(work);
743 for (auto& t : pool) t.join();
744 //: a start whose fit failed (a non-finite log posterior) never wins, wherever it sits in the list
745 auto score = [&](std::size_t i) { const double v = out.fits[i].pt.logpost; return std::isfinite(v) ? v : -std::numeric_limits<double>::infinity(); };
746 for (std::size_t i = 1; i < out.fits.size(); ++i)
747 if (score(i) > score(out.best)) out.best = i;
748 return out;
749}

790struct BayesianDecayLambdaOptions {
794 double prior_slope = 0.0;

816 int max_iter = 1000;
820 double tol = 1e-6;

857inline BayesianDecayLambdaGrid bayesian_decay_fit_lambda_grid(const BayesianDecayExperiment& f, const BayesianDecayTensors& env,
858 const std::vector<double>& log10_nodes, const std::vector<double>& theta_centre,
859 double log10_centre, const BayesianDecayLambdaOptions& opt) {
860 BayesianDecayLambdaGrid G;
861 std::vector<double> nodes = log10_nodes;
862 std::sort(nodes.begin(), nodes.end());
863 nodes.erase(std::unique(nodes.begin(), nodes.end()), nodes.end());
864 BayesianDecayExperiment g = f;
865 //: one node, fitted from `start`, inserted so that `G.nodes` stays sorted by lambda
866 auto fit_into = [&](double x, const std::vector<double>& start) {
867 BayesianDecayLambdaNode N;
868 N.log10_lam = x;
869 g.fixed_values["log10_lam"] = {x};
870 N.fit = bayesian_decay_fit_node(g, env, start, 1.0, opt.max_iter, opt.tol);
871 //: the node's evidence with the exact Hessian (bayesian_decay_hessian): Fisher's drops the
872 //: second-derivative term, 11.7 nats on CBM56, and the weights are differences of these
873 if (opt.exact_evidence) {
874 N.fit = bayesian_decay_polish_exact(g, env, N.fit);
875 N.fit.evidence = bayesian_decay_laplace_evidence_exact(g, env, N.fit.pt);
876 }
877 const std::vector<double> Sig = bayesian_decay_covariance(N.fit.pt, g.dim);
878 bayesian_decay_distribution_with_sd(g, env, N.fit.theta, Sig, N.p, N.p_sd, &N.mean_rel, &N.mean_rel_sd);
879 const auto at = std::lower_bound(G.nodes.begin(), G.nodes.end(), x,
880 [](const BayesianDecayLambdaNode& a, double b) { return a.log10_lam < b; });
881 G.nodes.insert(at, N);
882 };
883 //: the coarse pass: the node nearest the centre from `theta_centre`, the others warm-started from
884 //: their neighbour, walking outward
885 std::size_t c0 = 0;
886 for (std::size_t i = 0; i < nodes.size(); ++i)
887 if (std::fabs(nodes[i] - log10_centre) < std::fabs(nodes[c0] - log10_centre)) c0 = i;
888 fit_into(nodes[c0], theta_centre);
889 for (std::size_t i = c0 + 1; i < nodes.size(); ++i) fit_into(nodes[i], G.nodes.back().fit.theta);
890 for (std::size_t i = c0; i-- > 0;) fit_into(nodes[i], G.nodes.front().fit.theta);
891 G.n_coarse = G.nodes.size();
892
893 //: the tilted evidence, which is what the weights and the refinement window are read from
894 auto tilted = [&](const BayesianDecayLambdaNode& N) { return N.fit.evidence + opt.prior_slope * N.log10_lam; };
895 if (opt.refine_step > 0.0 && opt.refine_window > 0.0 && G.nodes.size() > 1) {
896 double best = -std::numeric_limits<double>::infinity();
897 double lo = 0.0, hi = 0.0;
898 bool any = false;
899 for (const auto& N : G.nodes)
900 if (std::isfinite(tilted(N)) && tilted(N) > best) best = tilted(N);
901 for (const auto& N : G.nodes)
902 if (std::isfinite(tilted(N)) && tilted(N) >= best - opt.refine_window) {
903 lo = any ? std::min(lo, N.log10_lam) : N.log10_lam;
904 hi = any ? std::max(hi, N.log10_lam) : N.log10_lam;
905 any = true;
906 }
907 if (any) {
908 //: half a decade beyond the kept nodes, inside the coarse grid's own range
909 lo = std::max(nodes.front(), lo - 0.5);
910 hi = std::min(nodes.back(), hi + 0.5);
911 std::vector<double> wanted;
912 for (double x = lo; x <= hi + 1e-9; x += opt.refine_step) {
913 bool have = false;
914 for (const auto& N : G.nodes) have = have || std::fabs(N.log10_lam - x) < 1e-9;
915 if (!have) wanted.push_back(x);
916 }
917 if (int(wanted.size()) > opt.max_refine) wanted.resize(std::size_t(opt.max_refine));
918 for (double x : wanted) {
919 //: warm start from the nearest node already fitted
920 std::size_t near = 0;
921 for (std::size_t i = 0; i < G.nodes.size(); ++i)
922 if (std::fabs(G.nodes[i].log10_lam - x) < std::fabs(G.nodes[near].log10_lam - x)) near = i;
923 fit_into(x, G.nodes[near].fit.theta);
924 }
925 }
926 }
927
928 //: the sweeps: a node fitted from its neighbour's end point may find a better mode than its own start did
929 for (int pass = 0; pass < opt.sweeps; ++pass) {
930 std::size_t improved = 0;
931 for (std::size_t i = 0; i < G.nodes.size(); ++i) {
932 for (int side = 0; side < 2; ++side) {
933 const std::size_t j = side == 0 ? (i == 0 ? i : i - 1) : (i + 1 < G.nodes.size() ? i + 1 : i);
934 if (j == i) continue;
935 const double before = G.nodes[i].fit.pt.logpost;
936 BayesianDecayLambdaNode trial;
937 trial.log10_lam = G.nodes[i].log10_lam;
938 g.fixed_values["log10_lam"] = {trial.log10_lam};
939 //: the optional screen (off by default): score the neighbour's end point at THIS node's lambda
940 //: first -- one evaluation against a whole fit -- and skip the trial unless it is already better.
941 //: It catches the cliff case the sweep exists for, but NOT every move: measured on CBM56 it kept
942 //: 9 of 18 and moved the answer (see the option's own note).
943 ++G.n_screened;
944 if (opt.sweep_screen && !(bayesian_decay_log_posterior(g, env, G.nodes[j].fit.theta) > before)) continue;
945 ++G.n_swept_fits;
946 trial.fit = bayesian_decay_fit_node(g, env, G.nodes[j].fit.theta, 1.0, opt.max_iter, opt.tol);
947 if (!(trial.fit.pt.logpost > before + 1e-6)) continue;
948 if (opt.exact_evidence) {
949 trial.fit = bayesian_decay_polish_exact(g, env, trial.fit);
950 trial.fit.evidence = bayesian_decay_laplace_evidence_exact(g, env, trial.fit.pt);
951 }
952 const std::vector<double> Sig = bayesian_decay_covariance(trial.fit.pt, g.dim);
953 bayesian_decay_distribution_with_sd(g, env, trial.fit.theta, Sig, trial.p, trial.p_sd, &trial.mean_rel, &trial.mean_rel_sd);
954 G.nodes[i] = trial;
955 ++improved;
956 }
957 }
958 G.n_improved += improved;
959 if (improved == 0) break;
960 }
961
962 //: Tierney-Kadane where the node carries weight (the weights are the evidences, so they are known
963 //: before any summary is formed): two tilted fits per node, the delta-method pair kept beside it
964 if (opt.summary_method == "tk") {
965 BayesianEvidenceMixture w_only;
966 for (const auto& N : G.nodes) w_only.add(tilted(N), 0.0, 0.0);
967 const std::vector<double> w = w_only.weights();
968 for (std::size_t i = 0; i < G.nodes.size(); ++i) {
969 if (!(w[i] >= opt.tk_min_weight)) continue;
970 g.fixed_values["log10_lam"] = {G.nodes[i].log10_lam};
971 const BayesianDecayTkSummary tk = bayesian_decay_tk_mean_rel(g, env, G.nodes[i].fit, opt.tk_eps, opt.tk_max_iter,
972 opt.tk_exact_curvature);
973 G.nodes[i].tk_mean = tk.mean; G.nodes[i].tk_sd = tk.sd; G.nodes[i].tk_ok = tk.ok;
974 G.nodes[i].tk_flat_curvature = tk.flat_curvature;
975 G.nodes[i].tk_flat_prior_share = tk.flat_prior_share;
976 for (int k = 0; k < 2; ++k) {
977 G.nodes[i].tk_shift[k] = tk.shift[k]; G.nodes[i].tk_dlogdet[k] = tk.dlogdet[k];
978 G.nodes[i].tk_flat_ratio[k] = tk.flat_ratio[k];
979 G.nodes[i].tk_flat_share[k] = tk.flat_share[k];
980 }
981 if (tk.ok) ++G.n_tk;
982 }
983 }
984
985 BayesianEvidenceMixture scalar, vec;
986 for (const auto& N : G.nodes) {
987 if (!std::isfinite(N.fit.evidence)) ++G.n_dropped;
988 //: a node TK did not reach (too little weight, or a tilted fit that did not converge) keeps the
989 //: delta method, which is what the prototype does
990 const bool use_tk = N.tk_ok && opt.summary_method == "tk";
991 scalar.add(tilted(N), use_tk ? N.tk_mean : N.mean_rel,
992 use_tk ? N.tk_sd * N.tk_sd : N.mean_rel_sd * N.mean_rel_sd);
993 std::vector<double> v(N.p_sd.size());
994 for (std::size_t j = 0; j < v.size(); ++j) v[j] = N.p_sd[j] * N.p_sd[j];
995 vec.add_vector(tilted(N), N.p, v);
996 }
997 G.weights = scalar.weights();
998 scalar.moments(&G.mean_rel, &G.mean_rel_sd);
999 vec.moments_vector(&G.p_mean, &G.p_sd);
1000 G.mean_rel_q16 = scalar.quantile(0.15865525393145705);
1001 G.mean_rel_q50 = scalar.quantile(0.5);
1002 G.mean_rel_q84 = scalar.quantile(0.8413447460685429);
1003 G.p_q16 = vec.quantile_vector(0.15865525393145705);
1004 G.p_q84 = vec.quantile_vector(0.8413447460685429);
1005 return G;
1006}
