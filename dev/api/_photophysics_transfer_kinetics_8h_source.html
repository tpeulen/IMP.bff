97template <typename T>
98inline TransferPair<T> transfer_pair_t(const T& tau_a, const T& tau_b, const T& k_ab,
99 const T& k_ba, const T& p_a, const T& p_b,
100 const T& m_a, const T& m_b) {
101 using std::sqrt;
102 TransferPair<T> q;
103 q.it_a = T(1.0) / tau_a;
104 q.it_b = T(1.0) / tau_b;
105 const T split = q.it_a - q.it_b + k_ab - k_ba;
106 q.delta = sqrt(split * split + T(4.0) * k_ab * k_ba);
107 q.l1 = T(0.5) * (-q.it_a - q.it_b - k_ab - k_ba + q.delta);
108 q.l2 = q.l1 - q.delta;
109 q.c = m_a * (p_a * (-q.l2 - q.it_a - k_ab) + p_b * k_ba) +
110 m_b * (p_a * k_ab + p_b * (-q.l2 - q.it_b - k_ba));
111 q.s = m_a * p_a + m_b * p_b;
112 return q;
113}

118template <typename T>
119inline void transfer_pair_components_t(const T& tau_a, const T& tau_b, const T& k_ab,
120 const T& k_ba, const T& p_a, const T& p_b,
121 const T& m_a, const T& m_b, int mode, double eps,
122 TransferComponent<T> out[2]) {
123 const TransferPair<T> q = transfer_pair_t(tau_a, tau_b, k_ab, k_ba, p_a, p_b, m_a, m_b);
124 if (mode == TRANSFER_CHISURF_REGULARISED) {
125 const T common = T(1.0) / (q.delta + T(1e-9));
126 out[0] = {common * q.c, -q.l1, TRANSFER_EXPONENTIAL};
127 out[1] = {common * (m_a * (p_a * (q.l1 + q.it_a + k_ab) - p_b * k_ba) +
128 m_b * (-p_a * k_ab + p_b * (q.l1 + q.it_b + k_ba))),
129 -q.l2, TRANSFER_EXPONENTIAL};
130 return;
131 }
132 if (tttrlib::ad_value(q.delta) > eps * std::fabs(tttrlib::ad_value(q.l1))) {
133 const T a1 = q.c / q.delta;
134 out[0] = {a1, -q.l1, TRANSFER_EXPONENTIAL};
135 out[1] = {q.s - a1, -q.l2, TRANSFER_EXPONENTIAL};
136 return;
137 }
138 const T l = T(0.5) * (q.l1 + q.l2);
139 out[0] = {q.s, -l, TRANSFER_EXPONENTIAL};
140 out[1] = {q.c, -l, TRANSFER_T_EXPONENTIAL};
141}

144inline double transfer_phi(double delta, double t) {
145 const double x = delta * t;
146 return x == 0.0 ? t : std::expm1(x) / delta;
147}
148template <typename G>
149inline tttrlib::Dual<G> transfer_phi(const tttrlib::Dual<G>& delta, double t) {
150 const double x = delta.val * t;
151 // d phi / d Delta = t^2 (x e^x - expm1(x)) / x^2, which is t^2/2 at 0.
152 const double slope = x == 0.0 ? 0.5 * t * t
153 : t * t * (x * std::exp(x) - std::expm1(x)) / (x * x);
154 tttrlib::Dual<G> r(transfer_phi(delta.val, t), delta.grad);
155 r.grad *= slope;
156 return r;
157}

160template <typename T>
161inline T transfer_pair_decay_t(double t, const T& tau_a, const T& tau_b, const T& k_ab,
162 const T& k_ba, const T& p_a, const T& p_b, const T& m_a,
163 const T& m_b) {
164 using std::exp;
165 const TransferPair<T> q = transfer_pair_t(tau_a, tau_b, k_ab, k_ba, p_a, p_b, m_a, m_b);
166 return exp(q.l2 * T(t)) * (q.s + q.c * transfer_phi(q.delta, t));
167}
