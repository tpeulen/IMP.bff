784 std::vector<int> get_neighbor_idx_offsets(double neighbor_radius = -1){
785 if(neighbor_radius < 0){
786 neighbor_radius = get_path_map_header().get_neighbor_radius();
787 }
788 const int nn = ceil(neighbor_radius);
789 const double nr2 = neighbor_radius * neighbor_radius;
790
791 const GridHeader* header = get_header();
792 int nx = header->get_nx();
793 int ny = header->get_ny();
794 int nx_ny = nx * ny;
795
796 // FPS's link search (set_cube_stencil): every offset of the
797 // (2n+1)^3 cube but the tile itself, at its Euclidean length --
798 // `av_routines.cpp:168-205`, `sqrts_dg`. The radius plays no part.
799 if(cube_stencil_ > 0){
800 const int n = cube_stencil_;
801 std::vector<int> offsets;
802 offsets.reserve((size_t) 5 * ((2*n + 1) * (2*n + 1) * (2*n + 1) - 1));
803 for(int z = -n; z <= n; z++)
804 for(int y = -n; y <= n; y++)
805 for(int x = -n; x <= n; x++){
806 const int d2 = z*z + y*y + x*x;
807 if(d2 == 0) continue;
808 int d;
809 const float len = std::sqrt((float) d2);
810 std::memcpy(&d, &len, sizeof(float));
811 offsets.emplace_back(z);
812 offsets.emplace_back(y);
813 offsets.emplace_back(x);
814 offsets.emplace_back(z * nx_ny + y * nx + x);
815 offsets.emplace_back(d);
816 }
817 return offsets;
818 }
819
820 // Historical stencil: the loops run -nn <= z < nn, so the +nn face is
821 // missing (a length-2 jump exists towards -x/-y/-z only) and the
822 // tile itself is included. The symmetric stencil (see
823 // set_symmetric_stencil) runs -nn..nn and drops the self offset.
824 const int hi = symmetric_stencil_ ? nn : nn - 1;
825 std::vector<int> offsets;
826 for(int z = -nn; z <= hi; z += 1) {
827 double dz2 = z * z;
828 int oz = z * nx_ny;
829 for(int y = -nn; y <= hi; y++) {
830 int oy = y * nx;
831 double dz2_dy2 = dz2 + y * y;
832 for(int x = -nn; x <= hi; x++) {
833 int ox = x;
834 int dz2_dy2_dx2 = dz2_dy2 + x * x;
835 if(symmetric_stencil_ && dz2_dy2_dx2 == 0) continue;
836 if(dz2_dy2_dx2 <= nr2){
837 int d; // edge_cost is a float stored in an 32bit int
838 float *p = (float*) &d; // Make a float pointer point at the integer
839 *p = sqrt((float) dz2_dy2_dx2); // Pretend that the integer is a float and store the value
840 int tile_offset = oz + oy + ox;
841 offsets.emplace_back(z);
842 offsets.emplace_back(y);
843 offsets.emplace_back(x);
844 offsets.emplace_back(tile_offset);
845 offsets.emplace_back(d);
846 }
847 }
848 }
849 }
850 return offsets;
851 }

1437template <typename T>
1438inline void drop_source_obstruction(T *counts, const IMP::algebra::Vector3D &origin,
1439 int n, double spacing,
1440 const IMP::algebra::Vector3D &c,
1441 double source_radius, double extra) {
1442 /* Nothing to subtract when the attachment atom contributed nothing. A
1443 radius of zero means it is not an obstacle -- the array door builds its
1444 obstacles from a caller's list and the source is not in it -- and
1445 subtracting a sphere that was never added would open voxels no rule
1446 opened. */
1447 if (source_radius <= 0.0) return;
1448 const double radius = source_radius + extra;
1449 if (radius <= 0.0) return;
1450 const double r2 = radius * radius;
1451 const long nxy = (long) n * n;
1452 for (int iz = 0; iz < n; iz++) {
1453 const double dz = origin[2] + iz * spacing - c[2];
1454 const double dz2 = dz * dz;
1455 if (dz2 >= r2) continue;
1456 for (int iy = 0; iy < n; iy++) {
1457 const double dy = origin[1] + iy * spacing - c[1];
1458 const double dyz2 = dz2 + dy * dy;
1459 if (dyz2 >= r2) continue;
1460 const long row = (long) iz * nxy + (long) iy * n;
1461 for (int ix = 0; ix < n; ix++) {
1462 const double dx = origin[0] + ix * spacing - c[0];
1463 // strict `<`, as ProbeAccessibleVolumeOccupancyMap::add_sphere tests
1464 if (dyz2 + dx * dx < r2) {
1465 T &v = counts[row + ix];
1466 if (v > 0) v -= 1;
1467 }
1468 }
1469 }
1470 }
1471}
