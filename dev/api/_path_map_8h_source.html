767 std::vector<int> get_neighbor_idx_offsets(double neighbor_radius = -1){
768 if(neighbor_radius < 0){
769 neighbor_radius = get_path_map_header().get_neighbor_radius();
770 }
771 const int nn = ceil(neighbor_radius);
772 const double nr2 = neighbor_radius * neighbor_radius;
773
774 const GridHeader* header = get_header();
775 int nx = header->get_nx();
776 int ny = header->get_ny();
777 int nx_ny = nx * ny;
778
779 // Historical stencil: the loops run -nn <= z < nn, so the +nn face is
780 // missing (a length-2 jump exists towards -x/-y/-z only) and the
781 // tile itself is included. The symmetric stencil (see
782 // set_symmetric_stencil) runs -nn..nn and drops the self offset.
783 const int hi = symmetric_stencil_ ? nn : nn - 1;
784 std::vector<int> offsets;
785 for(int z = -nn; z <= hi; z += 1) {
786 double dz2 = z * z;
787 int oz = z * nx_ny;
788 for(int y = -nn; y <= hi; y++) {
789 int oy = y * nx;
790 double dz2_dy2 = dz2 + y * y;
791 for(int x = -nn; x <= hi; x++) {
792 int ox = x;
793 int dz2_dy2_dx2 = dz2_dy2 + x * x;
794 if(symmetric_stencil_ && dz2_dy2_dx2 == 0) continue;
795 if(dz2_dy2_dx2 <= nr2){
796 int d; // edge_cost is a float stored in an 32bit int
797 float *p = (float*) &d; // Make a float pointer point at the integer
798 *p = sqrt((float) dz2_dy2_dx2); // Pretend that the integer is a float and store the value
799 int tile_offset = oz + oy + ox;
800 offsets.emplace_back(z);
801 offsets.emplace_back(y);
802 offsets.emplace_back(x);
803 offsets.emplace_back(tile_offset);
804 offsets.emplace_back(d);
805 }
806 }
807 }
808 }
809 return offsets;
810 }

1334template <typename T>
1335inline void drop_source_obstruction(T *counts, const IMP::algebra::Vector3D &origin,
1336 int n, double spacing,
1337 const IMP::algebra::Vector3D &c,
1338 double source_radius, double extra) {
1339 /* Nothing to subtract when the attachment atom contributed nothing. A
1340 radius of zero means it is not an obstacle -- the array door builds its
1341 obstacles from a caller's list and the source is not in it -- and
1342 subtracting a sphere that was never added would open voxels no rule
1343 opened. */
1344 if (source_radius <= 0.0) return;
1345 const double radius = source_radius + extra;
1346 if (radius <= 0.0) return;
1347 const double r2 = radius * radius;
1348 const long nxy = (long) n * n;
1349 for (int iz = 0; iz < n; iz++) {
1350 const double dz = origin[2] + iz * spacing - c[2];
1351 const double dz2 = dz * dz;
1352 if (dz2 >= r2) continue;
1353 for (int iy = 0; iy < n; iy++) {
1354 const double dy = origin[1] + iy * spacing - c[1];
