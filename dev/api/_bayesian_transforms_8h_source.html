80template <typename T = double>
81struct BayesianLogitTransform {
82 double lo = 0.0, hi = 1.0;
83 BayesianLogitTransform() = default;
84 BayesianLogitTransform(double lo_, double hi_) : lo(lo_), hi(hi_) {
85 if (!(hi_ > lo_)) throw std::invalid_argument("BayesianLogitTransform: hi must exceed lo");
86 }
87 static T log_sigmoid(const T& z) {
88 using std::exp; using std::log1p;
89 return (z > T(0.0)) ? -log1p(exp(-z)) : z - log1p(exp(z));
90 }
91 void to_constrained(const T* z, std::size_t n, T* x) const {
92 using std::exp;
93 for (std::size_t i = 0; i < n; ++i) {
94 const T s = (z[i] > T(0.0)) ? T(1.0) / (T(1.0) + exp(-z[i]))
95 : exp(z[i]) / (T(1.0) + exp(z[i]));
96 x[i] = T(lo) + T(hi - lo) * s;
97 }
98 }
99 void to_unconstrained(const T* x, std::size_t n, T* z) const {
100 using std::log;
101 for (std::size_t i = 0; i < n; ++i) {
102 T p = (x[i] - T(lo)) / T(hi - lo);
103 if (p < T(1e-12)) p = T(1e-12);
104 if (p > T(1.0 - 1e-12)) p = T(1.0 - 1e-12);
105 z[i] = log(p) - log(T(1.0) - p);
106 }
107 }
108 T log_abs_det(const T* z, std::size_t n) const {
109 using std::log;
110 T s = T(0.0);
111 for (std::size_t i = 0; i < n; ++i)
112 s += T(std::log(hi - lo)) + log_sigmoid(z[i]) + log_sigmoid(-z[i]);
113 return s;
114 }
115};

125template <typename T = double>
126struct BayesianALRTransform {
127 std::size_t n = 0;
128 BayesianALRTransform() = default;
129 explicit BayesianALRTransform(std::size_t n_) : n(n_) {
130 if (n_ < 2) throw std::invalid_argument("BayesianALRTransform: need at least two weights");
131 }
133 void to_constrained(const T* z, std::size_t, T* x) const {
134 using std::exp; using std::log;
135 T m = T(0.0); // max([z, 0]) for stability
136 for (std::size_t i = 0; i + 1 < n; ++i) if (z[i] > m) m = z[i];
137 T s = T(0.0);
138 for (std::size_t i = 0; i + 1 < n; ++i) { x[i] = exp(z[i] - m); s += x[i]; }
139 x[n - 1] = exp(-m); s += x[n - 1];
140 for (std::size_t i = 0; i < n; ++i) x[i] /= s;
141 }
142 void to_unconstrained(const T* x, std::size_t, T* z) const {
143 using std::log;
144 for (std::size_t i = 0; i + 1 < n; ++i) z[i] = log(x[i]) - log(x[n - 1]);
145 }
146 T log_abs_det(const T*, std::size_t) const { return T(0.0); }
147};

170struct BayesianSumToZeroTransform {
171 std::size_t n = 0;
172 std::vector<double> Q;
173 BayesianSumToZeroTransform() = default;
174 explicit BayesianSumToZeroTransform(std::size_t n_) : n(n_), Q(helmert_basis(n_)) {}
175
177 static std::vector<double> helmert_basis(std::size_t n) {
178 if (n < 2) throw std::invalid_argument("helmert_basis: need n >= 2");
179 std::vector<double> q(n * (n - 1), 0.0);
180 for (std::size_t j = 1; j < n; ++j) {
181 const double s = std::sqrt(double(j) * (double(j) + 1.0));
182 for (std::size_t i = 0; i < j; ++i) q[i * (n - 1) + (j - 1)] = 1.0 / s;
183 q[j * (n - 1) + (j - 1)] = -double(j) / s;
184 }
185 return q;
186 }
188 void to_constrained(const T* z, std::size_t, T* c) const {
189 for (std::size_t i = 0; i < n; ++i) {
190 T s = T(0.0);
191 for (std::size_t j = 0; j + 1 < n; ++j) s += T(Q[i * (n - 1) + j]) * z[j];
192 c[i] = s;
193 }
194 }
195 void to_unconstrained(const T* c, std::size_t, T* z) const {
196 for (std::size_t j = 0; j + 1 < n; ++j) {
197 T s = T(0.0);
198 for (std::size_t i = 0; i < n; ++i) s += T(Q[i * (n - 1) + j]) * c[i];
199 z[j] = s;
200 }
201 }
202 T log_abs_det(const T*, std::size_t) const { return T(0.0); }
203};
