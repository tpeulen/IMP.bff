94 bool add(std::size_t offset, std::size_t size, BayesianPriorFamily family,
95 BayesianTransformKind transform, double a, double b) {
96 const bool gaussian_in_z =
97 (family == BAYESIAN_PRIOR_GAUSSIAN_ON_Z) ||
98 (family == BAYESIAN_PRIOR_GAUSSIAN && transform == BAYESIAN_TRANSFORM_IDENTITY) ||
99 (family == BAYESIAN_PRIOR_LOGNORMAL && transform == BAYESIAN_TRANSFORM_LOG);
100 if (gaussian_in_z) {
101 if (!(b > 0.0)) throw std::invalid_argument("BayesianTransformedGaussianPrior: sd must be positive");
102 const double mu = (family == BAYESIAN_PRIOR_LOGNORMAL) ? std::log(a) : a;
103 for (std::size_t i = 0; i < size; ++i) {
104 idx_.push_back(offset + i);
105 mu_.push_back(mu);
106 sd_.push_back(b);
107 }
108 return true;
109 }
110 if (family == BAYESIAN_PRIOR_UNIFORM && transform == BAYESIAN_TRANSFORM_LOGIT) {
111 // The flat density contributes -log(hi - lo) per element and the
112 // logit's Jacobian contributes +log(hi - lo) per element, so when the
113 // transform's bounds ARE the prior's bounds -- which is the only case
114 // this overload describes, since it takes one pair -- the two cancel
115 // exactly and what survives is the shape term alone. Adding either one
116 // without the other is the mistake this comment exists to prevent.
117 if (!(b > a)) throw std::invalid_argument("BayesianTransformedGaussianPrior: need hi > lo");
118 for (std::size_t i = 0; i < size; ++i) logit_.push_back(offset + i);
119 return true;
120 }
121 return false;
122 }

138 bool add_vector(std::size_t offset, std::size_t size, BayesianPriorFamily family,
139 BayesianTransformKind transform, const double* a, const double* b) {
140 const bool gaussian_in_z =
141 (family == BAYESIAN_PRIOR_GAUSSIAN_ON_Z) ||
142 (family == BAYESIAN_PRIOR_GAUSSIAN && transform == BAYESIAN_TRANSFORM_IDENTITY) ||
143 (family == BAYESIAN_PRIOR_LOGNORMAL && transform == BAYESIAN_TRANSFORM_LOG);
144 if (!gaussian_in_z) return false;
145 for (std::size_t i = 0; i < size; ++i) {
146 if (!(b[i] > 0.0)) throw std::invalid_argument("BayesianTransformedGaussianPrior: sd must be positive");
147 idx_.push_back(offset + i);
148 mu_.push_back((family == BAYESIAN_PRIOR_LOGNORMAL) ? std::log(a[i]) : a[i]);
149 sd_.push_back(b[i]);
150 }
151 return true;
152 }

160 T log_prob(const T* z) const {
161 T s = T(0.0);
162 for (std::size_t k = 0; k < idx_.size(); ++k) {
163 const T d = (z[idx_[k]] - T(mu_[k])) / T(sd_[k]);
164 s -= T(0.5) * d * d;
165 }
166 for (std::size_t k = 0; k < logit_.size(); ++k) {
167 const T& zz = z[logit_[k]];
168 s += BayesianLogitTransform<T>::log_sigmoid(zz) + BayesianLogitTransform<T>::log_sigmoid(-zz);
169 }
170 return s + T(normaliser());
171 }
172
174 void add_gradient(const T* z, T* out) const {
175 for (std::size_t k = 0; k < idx_.size(); ++k)
176 out[idx_[k]] -= (z[idx_[k]] - T(mu_[k])) / T(sd_[k] * sd_[k]);
177 for (std::size_t k = 0; k < logit_.size(); ++k) {
178 const T s = sigmoid(z[logit_[k]]);
179 out[logit_[k]] += T(1.0) - T(2.0) * s;
180 }
181 }
182
191 void add_hessian(const T* z, std::size_t n, T* out) const {
192 for (std::size_t k = 0; k < idx_.size(); ++k)
193 out[idx_[k] * n + idx_[k]] -= T(1.0 / (sd_[k] * sd_[k]));
194 for (std::size_t k = 0; k < logit_.size(); ++k) {
195 const T s = sigmoid(z[logit_[k]]);
196 out[logit_[k] * n + logit_[k]] -= T(2.0) * s * (T(1.0) - s);
197 }
198 }
199
200 private:
201 static T sigmoid(const T& z) {
202 using std::exp;
203 return (z > T(0.0)) ? T(1.0) / (T(1.0) + exp(-z)) : exp(z) / (T(1.0) + exp(z));
204 }
205 double normaliser() const {
206 double c = const_;
207 for (std::size_t k = 0; k < sd_.size(); ++k)
208 c += -std::log(sd_[k]) - 0.5 * std::log(2.0 * M_PI);
209 return c;
210 }
211
212 std::vector<std::size_t> idx_, logit_;
213 std::vector<double> mu_, sd_;
214 double const_ = 0.0;
215};
