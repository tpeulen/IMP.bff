44inline WarmupWindows warmup_windows(int n_warmup, int init_buffer = 75, int term_buffer = 50, int base_window = 25) {
45 WarmupWindows w;
46 w.init_buffer = init_buffer;
47 w.term_buffer = term_buffer;
48 if (n_warmup < 20) {
49 w.init_buffer = n_warmup;
50 w.term_buffer = 0;
51 return w;
52 }
53 if (init_buffer + base_window + term_buffer > n_warmup) {
54 w.init_buffer = int(std::lround(0.15 * n_warmup));
55 w.term_buffer = int(std::lround(0.10 * n_warmup));
56 base_window = n_warmup - w.init_buffer - w.term_buffer;
57 if (base_window < 2) {
58 w.init_buffer = n_warmup;
59 w.term_buffer = 0;
60 return w;
61 }
62 }
63 int start = w.init_buffer, window = base_window;
64 const int last = n_warmup - w.term_buffer;
65 while (start + window <= last) {
66 int end = start + window;
67 if (end + 2 * window > last) end = last;
68 w.ends.push_back(end);
69 start = end;
70 window *= 2;
71 }
72 return w;
73}

82 void restart(double step) { restart_log(std::log(10.0 * step)); }
84 void restart_log(double mu) {
85 mu_ = mu;
86 s_bar_ = 0.0;
87 x_bar_ = mu;
88 counter_ = 0.0;
89 }
91 double learn_log(double adapt_stat) {
92 counter_ += 1.0;
93 adapt_stat = std::min(1.0, adapt_stat);
94 const double eta = 1.0 / (counter_ + t0_);
95 s_bar_ = (1.0 - eta) * s_bar_ + eta * (delta_ - adapt_stat);
96 const double x = mu_ - s_bar_ * std::sqrt(counter_) / gamma_;
97 const double x_eta = std::pow(counter_, -kappa_);
98 x_bar_ = (1.0 - x_eta) * x_bar_ + x_eta * x;
99 return x;
100 }
102 double learn(double adapt_stat) { return std::exp(learn_log(adapt_stat)); }
104 double final_log() const { return x_bar_; }
106 double final_step(double current) const { return counter_ > 0.0 ? std::exp(x_bar_) : current; }
107 double target() const { return delta_; }
108 void set_target(double delta) { delta_ = delta; }
109
110 private:
111 double delta_, gamma_, t0_, kappa_;
112 double mu_ = 0.0, s_bar_ = 0.0, x_bar_ = 0.0, counter_ = 0.0;

122 dim_ = dim;
123 dense_ = dense;
124 restart();
125 }
126 void restart() {
127 n_ = 0.0;
128 m_.assign(dim_, 0.0);
129 m2_.assign(dense_ ? dim_ * dim_ : dim_, 0.0);
130 }
131 void add(const std::vector<double>& q) {
132 n_ += 1.0;
133 std::vector<double> delta(dim_);
134 for (std::size_t i = 0; i < dim_; ++i) {
135 delta[i] = q[i] - m_[i];
136 m_[i] += delta[i] / n_;
137 }
138 if (dense_) {
139 for (std::size_t i = 0; i < dim_; ++i)
140 for (std::size_t j = 0; j < dim_; ++j) m2_[i * dim_ + j] += (q[i] - m_[i]) * delta[j];
141 } else {
142 for (std::size_t i = 0; i < dim_; ++i) m2_[i] += (q[i] - m_[i]) * delta[i];
143 }
144 }
145 double n() const { return n_; }
146 bool dense() const { return dense_; }
149 std::vector<double> regularised() const {
150 std::vector<double> out(m2_.size(), 0.0);
151 const double w = n_ / (n_ + 5.0), r = 1e-3 * (5.0 / (n_ + 5.0));
152 for (std::size_t k = 0; k < out.size(); ++k) out[k] = n_ > 1.0 ? w * m2_[k] / (n_ - 1.0) : 0.0;
153 if (dense_) {
154 for (std::size_t i = 0; i < dim_; ++i) out[i * dim_ + i] += r;
155 } else {
156 for (double& v : out) v += r;
157 }
158 return out;
159 }
160
161 private:
162 std::size_t dim_ = 0;
163 bool dense_ = true;
164 double n_ = 0.0;
165 std::vector<double> m_, m2_;
166};
