101inline SamplingTarget with_finite_difference_gradient(SamplingTarget t, double h = 1e-6) {
102 const auto f = t.log_density;
103 const std::size_t n = t.dim;
104 t.log_density_gradient = [f, n, h](const std::vector<double>& x, std::vector<double>& g) {
105 g.assign(n, 0.0);
106 std::vector<double> y = x;
107 for (std::size_t i = 0; i < n; ++i) {
108 y[i] = x[i] + h;
109 const double up = f(y);
110 y[i] = x[i] - h;
111 const double dn = f(y);
112 y[i] = x[i];
113 g[i] = (up - dn) / (2.0 * h);
114 }
115 return f(x);
116 };
117 return t;
118}

207inline std::uint64_t sampler_split_seed(std::uint64_t base, std::uint64_t index) {
208 std::uint64_t z = base + 0x9E3779B97F4A7C15ull * (index + 1);
209 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ull;
210 z = (z ^ (z >> 27)) * 0x94D049BB133111EBull;
211 return z ^ (z >> 31);
212}

219inline SampleResult run_sampler(const SamplingTarget& target, const SamplerKernel& prototype,
220 const std::vector<std::vector<std::vector<double>>>& starts,
221 const SamplerOptions& opt = SamplerOptions()) {
222 if (starts.empty()) throw SamplerConfigurationError("run_sampler: no starts");
223 if (prototype.needs_gradient() && !target.has_gradient())
224 throw SamplerConfigurationError("sampler '" + prototype.name() + "' needs a gradient; the target has none "
225 "(with_finite_difference_gradient() gives it one explicitly)");
226 if (!target.log_density) throw SamplerConfigurationError("run_sampler: the target has no log density");
227 if (opt.draws < 0 || opt.warmup < 0 || opt.thin < 1) throw SamplerConfigurationError("run_sampler: warmup, draws >= 0 and thin >= 1");
228 const std::size_t G = starts.size();
229 for (const auto& s : starts) {
230 if (s.empty()) throw SamplerConfigurationError("run_sampler: a start without walkers");
231 if (!prototype.is_ensemble() && s.size() != 1)
232 throw SamplerConfigurationError("run_sampler: a chain kernel takes one walker per start");
233 for (const auto& w : s)
234 if (w.size() != target.dim) throw SamplerConfigurationError("run_sampler: a start of the wrong dimension");
235 }
236 struct Group {
237 std::vector<std::vector<std::vector<double>>> draws;
238 std::vector<std::vector<double>> lp;
239 std::vector<std::vector<std::vector<double>>> stats;
240 std::vector<std::vector<double>> warmup_sums;
241 std::string tuning;
242 long evaluations = 0;
243 std::string error;
244 };
245 std::vector<Group> groups(G);
246 const auto t0 = std::chrono::steady_clock::now();
247 auto run_group = [&](std::size_t g) {
248 try {
249 std::mt19937_64 rng(sampler_split_seed(opt.seed, g));
250 std::unique_ptr<SamplerKernel> k = prototype.clone();
251 k->initialize(target, starts[g], rng);
252 const std::size_t W = k->walkers().size();
253 const std::size_t S = k->stat_names().size();
254 Group& out = groups[g];
255 out.draws.assign(W, {});
256 out.lp.assign(W, {});
257 out.stats.assign(W, {});
258 out.warmup_sums.assign(W, std::vector<double>(S, 0.0));
259 std::vector<double> st;
260 k->begin_warmup(opt.warmup);
261 for (int i = 0; i < opt.warmup; ++i) {
262 k->transition(rng);
263 k->stats(st);
264 for (std::size_t w = 0; w < W; ++w)
265 for (std::size_t j = 0; j < S && (w + 1) * S <= st.size(); ++j) out.warmup_sums[w][j] += st[w * S + j];
266 }
267 k->end_warmup();
268 for (int i = 0; i < opt.draws * opt.thin; ++i) {
269 k->transition(rng);
270 if ((i + 1) % opt.thin) continue;
271 k->stats(st);
272 for (std::size_t w = 0; w < W; ++w) {
273 out.draws[w].push_back(k->walkers()[w]);
274 out.lp[w].push_back(k->log_density()[w]);
275 out.stats[w].emplace_back(st.begin() + w * S, st.begin() + (w + 1) * S);
276 }
277 if (g == 0 && opt.observer) opt.observer((i + 1) / opt.thin, opt.draws);
278 }
279 out.evaluations = k->evaluations();
280 out.tuning = k->tuning_json();
281 } catch (const std::exception& e) {
282 groups[g].error = e.what();
283 }
284 };
285 const std::size_t T = opt.threads > 0 ? std::min<std::size_t>(opt.threads, G) : G;
286 if (T <= 1) {
287 for (std::size_t g = 0; g < G; ++g) run_group(g);
288 } else {
289 std::vector<std::thread> pool;
290 std::size_t next = 0;
291 std::mutex m;
292 for (std::size_t t = 0; t < T; ++t)
293 pool.emplace_back([&] {
294 // a chain thread evaluates its density inline rather than contending for the one decay pool
295 internal::bayesian_serial_here() = true;
296 for (;;) {
297 std::size_t g;
298 {
299 std::lock_guard<std::mutex> lock(m);
300 if (next >= G) return;
301 g = next++;
302 }
303 run_group(g);
304 }
305 });
306 for (auto& th : pool) th.join();
307 }
308 SampleResult res;
309 res.kernel = prototype.name();
310 res.names = target.names;
311 res.stat_names = prototype.stat_names();
312 for (std::size_t g = 0; g < G; ++g) {
313 if (!groups[g].error.empty()) throw SamplerConfigurationError("run_sampler: chain " + std::to_string(g) + ": " + groups[g].error);
314 for (std::size_t w = 0; w < groups[g].draws.size(); ++w) {
315 res.draws.push_back(std::move(groups[g].draws[w]));
316 res.log_density.push_back(std::move(groups[g].lp[w]));
317 res.stats.push_back(std::move(groups[g].stats[w]));
318 res.warmup_stat_sums.push_back(std::move(groups[g].warmup_sums[w]));
319 res.independent_group.push_back(static_cast<int>(g));
320 }
321 res.evaluations += groups[g].evaluations;
322 res.tuning.push_back(groups[g].tuning);
323 }
324 res.seconds = std::chrono::duration<double>(std::chrono::steady_clock::now() - t0).count();
325 return res;
326}
