38inline std::vector<double> bayesian_pspline_basis(std::size_t n_points, std::size_t n_coef, int degree = 3) {
39 if (n_coef <= std::size_t(degree)) throw std::invalid_argument("bayesian_pspline_basis: n_coef must exceed the degree");
40 const std::size_t n_int = n_coef - std::size_t(degree);
41 const double h = 1.0 / double(n_int);
42 const std::size_t n_knots = n_int + 2 * std::size_t(degree) + 1;
43 std::vector<double> knot(n_knots);
44 for (std::size_t i = 0; i < n_knots; ++i) knot[i] = (double(i) - double(degree)) * h;
45 std::vector<double> out(n_points * n_coef, 0.0), N(std::size_t(degree) + 1);
46 for (std::size_t r = 0; r < n_points; ++r) {
47 double x = n_points > 1 ? double(r) / double(n_points - 1) : 0.0;
48 x = std::min(std::max(x, knot[std::size_t(degree)]), knot[n_knots - std::size_t(degree) - 1] - 1e-12);
49 // the knot span: knot[mu] <= x < knot[mu + 1], mu in [degree, degree + n_int)
50 std::size_t mu = std::size_t(degree) + std::min(n_int - 1, std::size_t(std::floor((x - knot[std::size_t(degree)]) / h)));
51 while (mu > std::size_t(degree) && x < knot[mu]) --mu;
52 while (mu + 1 < std::size_t(degree) + n_int && x >= knot[mu + 1]) ++mu;
53 // Cox-de Boor, the triangular scheme: N[j] = B_{mu - degree + j, k}(x)
54 N.assign(std::size_t(degree) + 1, 0.0);
55 N[std::size_t(degree)] = 1.0;
56 for (int k = 1; k <= degree; ++k) {
57 for (int j = degree - k; j <= degree; ++j) {
58 const std::size_t i = mu - std::size_t(degree) + std::size_t(j); // basis function index
59 const double left = (j > degree - k) ? (x - knot[i]) / (knot[i + std::size_t(k)] - knot[i]) * N[std::size_t(j)] : 0.0;
60 const double right = (j < degree) ? (knot[i + std::size_t(k) + 1] - x) / (knot[i + std::size_t(k) + 1] - knot[i + 1]) * N[std::size_t(j) + 1] : 0.0;
61 N[std::size_t(j)] = left + right;
62 }
63 }
64 for (int j = 0; j <= degree; ++j) {
65 const std::size_t i = mu - std::size_t(degree) + std::size_t(j);
66 if (i < n_coef) out[r * n_coef + i] = N[std::size_t(j)];
67 }
68 }
69 return out;
70}

158 T log_prob(const T* c, const T& lambda) const {
159 std::vector<T> d(n_rows());
160 differences(c, d.data());
161 T rough = T(0.0);
162 if (student_) {
163 for (std::size_t k = 0; k < n_rows(); ++k)
164 rough -= T(0.5) * T(nu_ + 1.0) * log1p_(lambda * d[k] * d[k] / T(nu_));
165 } else {
166 T q = T(0.0);
167 for (std::size_t k = 0; k < n_rows(); ++k) q += d[k] * d[k];
168 rough = T(-0.5) * lambda * q;
169 }
170 const T tilt = dot(v_tilt_.data(), c);
171 T lp = T(0.5 * rank_) * log_(lambda) + rough - T(0.5) * (tilt / T(tilt_sd_)) * (tilt / T(tilt_sd_));
172 if (order_ == 3) {
173 const T qd = dot(v_quad_.data(), c);
174 lp -= T(0.5) * (qd / T(quad_sd_)) * (qd / T(quad_sd_));
175 }
176 return lp;
177 }

180 void gradient(const T* c, const T& lambda, T* out) const {
181 std::vector<T> d(n_rows());
182 differences(c, d.data());
183 std::vector<T> w(n_rows());
184 for (std::size_t k = 0; k < n_rows(); ++k) {
185 if (student_) {
186 // d/dx of -0.5 (nu+1) log1p(lambda x^2 / nu)
187 const T den = T(1.0) + lambda * d[k] * d[k] / T(nu_);
188 w[k] = -T(nu_ + 1.0) * lambda * d[k] / T(nu_) / den;
189 } else {
190 w[k] = -lambda * d[k];
191 }
192 }
193 for (std::size_t i = 0; i < n_; ++i) out[i] = T(0.0);
194 for (std::size_t k = 0; k < n_rows(); ++k) {
195 const double* row = &D_[k * n_];
196 for (std::size_t i = 0; i < n_; ++i)
197 if (row[i] != 0.0) out[i] += w[k] * T(row[i]);
198 }
199 const T tilt = dot(v_tilt_.data(), c) / T(tilt_sd_ * tilt_sd_);
200 for (std::size_t i = 0; i < n_; ++i) out[i] -= tilt * T(v_tilt_[i]);
201 if (order_ == 3) {
202 const T qd = dot(v_quad_.data(), c) / T(quad_sd_ * quad_sd_);
203 for (std::size_t i = 0; i < n_; ++i) out[i] -= qd * T(v_quad_[i]);
204 }
205 }

215 void hessian(const T* c, const T& lambda, T* out) const {
216 for (std::size_t i = 0; i < n_ * n_; ++i) out[i] = T(0.0);
217 std::vector<T> d(n_rows());
218 if (student_) differences(c, d.data());
219 for (std::size_t k = 0; k < n_rows(); ++k) {
220 T s;
221 if (student_) {
222 const T u = lambda * d[k] * d[k] / T(nu_);
223 const T den = T(1.0) + u;
224 // d2/dx2 of -0.5 (nu+1) log1p(lambda x^2/nu)
225 s = -T(nu_ + 1.0) * lambda / T(nu_) * (T(1.0) - u) / (den * den);
226 } else {
227 s = -lambda;
228 }
229 const double* row = &D_[k * n_];
230 for (std::size_t i = 0; i < n_; ++i) {
231 if (row[i] == 0.0) continue;
232 for (std::size_t j = 0; j < n_; ++j)
233 if (row[j] != 0.0) out[i * n_ + j] += s * T(row[i] * row[j]);
234 }
235 }
236 add_outer(out, v_tilt_, 1.0 / (tilt_sd_ * tilt_sd_));
237 if (order_ == 3) add_outer(out, v_quad_, 1.0 / (quad_sd_ * quad_sd_));
238 }

241 void differences(const T* c, T* out) const {
242 for (std::size_t k = 0; k < n_rows(); ++k) {
243 const double* row = &D_[k * n_];
244 T s = T(0.0);
245 for (std::size_t i = 0; i < n_; ++i)
246 if (row[i] != 0.0) s += T(row[i]) * c[i];
247 out[k] = s;
248 }
249 }

269 void build_difference() {
270 D_.assign(n_rows() * n_, 0.0);
271 std::vector<double> coef(static_cast<std::size_t>(order_) + 1, 0.0);
272 double b = 1.0;
273 for (int j = 0; j <= order_; ++j) {
274 coef[static_cast<std::size_t>(j)] = ((j % 2) ? -b : b);
275 b = b * double(order_ - j) / double(j + 1);
276 }
277 // numpy's diff puts the newest sample first: row k is
278 // sum_j (-1)^j C(d, j) c[k + d - j]
279 for (std::size_t k = 0; k < n_rows(); ++k)
280 for (int j = 0; j <= order_; ++j)
281 D_[k * n_ + k + static_cast<std::size_t>(order_ - j)] = coef[static_cast<std::size_t>(j)];
282 }

286 void build_null_space() {
287 std::vector<double> v(n_);
288 const double mid = 0.5 * (double(n_) - 1.0);
289 double nv = 0.0;
290 for (std::size_t i = 0; i < n_; ++i) { v[i] = double(i) - mid; nv += v[i] * v[i]; }
291 nv = std::sqrt(nv);
292 v_tilt_.resize(n_);
293 for (std::size_t i = 0; i < n_; ++i) v_tilt_[i] = v[i] / nv;
294
295 std::vector<double> q(n_);
296 double mean = 0.0;
297 for (std::size_t i = 0; i < n_; ++i) { q[i] = v[i] * v[i]; mean += q[i]; }
298 mean /= double(n_);
299 double proj = 0.0;
300 for (std::size_t i = 0; i < n_; ++i) { q[i] -= mean; proj += q[i] * v_tilt_[i]; }
301 double nq = 0.0;
302 for (std::size_t i = 0; i < n_; ++i) { q[i] -= proj * v_tilt_[i]; nq += q[i] * q[i]; }
303 nq = std::sqrt(nq);
304 v_quad_.resize(n_);
305 for (std::size_t i = 0; i < n_; ++i) v_quad_[i] = (nq > 0.0 ? q[i] / nq : 0.0);
306 }
