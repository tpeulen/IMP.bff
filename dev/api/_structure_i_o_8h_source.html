40struct IMPBFFEXPORT AtomBond {
41 int a, b;
42 AtomBond(int a = 0, int b = 0) : a(a < b ? a : b), b(a < b ? b : a) {}
43 bool operator<(const AtomBond& o) const {
44 return a != o.a ? a < o.a : b < o.b;
45 }
46 bool operator==(const AtomBond& o) const { return a == o.a && b == o.b; }
47 IMP_SHOWABLE_INLINE(AtomBond, out << "AtomBond(" << a << ", " << b << ")");
48};
