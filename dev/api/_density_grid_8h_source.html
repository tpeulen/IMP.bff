231 void calc_all_voxel2loc() {
232 if (loc_calculated_) return;
233 const long nvox = get_number_of_voxels();
234 x_loc_.reset(new float[nvox]);
235 y_loc_.reset(new float[nvox]);
236 z_loc_.reset(new float[nvox]);
237 loc_n_ = nvox;
238 const float sp = header_.get_spacing();
239 const float ox = header_.get_xorigin(), oy = header_.get_yorigin(),
240 oz = header_.get_zorigin();
241 const int nx = header_.get_nx(), ny = header_.get_ny();
242 int ix = 0, iy = 0, iz = 0;
243 for (long ii = 0; ii < nvox; ++ii) {
244 x_loc_[ii] = ix * sp + ox;
245 y_loc_[ii] = iy * sp + oy;
246 z_loc_[ii] = iz * sp + oz;
247 if (++ix == nx) {
248 ix = 0;
249 if (++iy == ny) {
250 iy = 0;
251 ++iz;
252 }
253 }
254 }
255 loc_calculated_ = true;

276 int get_dim_index_by_location(float loc, int dim) const {
277 // The inputs are float, as IMP's are, and the arithmetic is double,
278 // as IMP's is: `DensityMap::get_dim_index_by_location` subtracts a
279 // float `loc_val` from `get_origin()[ind]`, which is a *double* built
280 // from the float origin, and divides by the float spacing -- every
281 // step promoted. Doing the subtraction and division in float instead
282 // rounds twice more, and at a half-voxel boundary that is a different
283 // voxel: the legacy lattice search seeds itself with
284 // `get_voxel_by_location(source)`, so a different voxel there is a
285 // different path and an AV tens of angstroms away.
286 const double o = dim == 0 ? header_.get_xorigin()
287 : (dim == 1 ? header_.get_yorigin()
288 : header_.get_zorigin());
289 const double sp = header_.get_spacing();
290 return static_cast<int>(std::floor(0.5 + (static_cast<double>(loc) - o) / sp));
291 }

319 void resample() {
320 reset_data();
321 calc_all_voxel2loc();
322 const long nvox_check = get_number_of_voxels();
323 IMP_USAGE_CHECK(static_cast<long>(data_.size()) == nvox_check,
324 "DensityGrid::resample: data is " << data_.size()
325 << " voxels, the header says " << nvox_check);
326 IMP_USAGE_CHECK(loc_n_ == nvox_check,
327 "DensityGrid::resample: location caches are for "
328 << loc_n_ << " voxels, the header says " << nvox_check);
329 const int nx = header_.get_nx(), ny = header_.get_ny(),
330 nz = header_.get_nz();
331 const long nxny = static_cast<long>(nx) * ny;
332 const float sp = header_.get_spacing();
333 const float ox = header_.get_xorigin(), oy = header_.get_yorigin(),
334 oz = header_.get_zorigin();
335 for (std::size_t ii = 0; ii < xyzr_.size(); ++ii) {
336 const IMP::algebra::Vector3D c = xyzr_[ii].get_coordinates();
337 const double r = xyzr_[ii].get_radius();
338 if (r <= 0.0) continue; // a zero radius blocks nothing
339 const double r2 = r * r;
340 int iminx, iminy, iminz, imaxx, imaxy, imaxz;
341 local_bounding_box(c, r, sp, ox, oy, oz, nx, ny, nz, &iminx,
342 &iminy, &iminz, &imaxx, &imaxy, &imaxz);
343 for (int iz = iminz; iz <= imaxz; ++iz) {
344 const long znxny = iz * nxny;
345 for (int iy = iminy; iy <= imaxy; ++iy) {
346 long ivox = znxny + static_cast<long>(iy) * nx + iminx;
347 for (int ix = iminx; ix <= imaxx; ++ix, ++ivox) {
348 const double dx = x_loc_[ivox] - c[0];
349 const double dy = y_loc_[ivox] - c[1];
350 const double dz = z_loc_[ivox] - c[2];
351 if (dx * dx + dy * dy + dz * dz < r2) data_[ivox] += 1.0;
352 }
353 }
354 }
355 }

372 static int lower_shift(double loc, float kdist, float orig, float sp,
373 int ndim) {
374 int i = static_cast<int>(std::floor((loc - kdist - orig) / sp));
375 if (i < 0) i = 0;

379 static int upper_shift(double loc, float kdist, float orig, float sp,
380 int ndim) {
381 int i = static_cast<int>(std::floor((loc + kdist - orig) / sp));
382 if (i < 0) i = 0;
383 if (i > ndim - 1) i = ndim - 1;
384 return i;
385 }

388 static void local_bounding_box(const IMP::algebra::Vector3D& c, double r,
389 float sp, float ox, float oy, float oz,
390 int nx, int ny, int nz, int* iminx,
391 int* iminy, int* iminz, int* imaxx,
392 int* imaxy, int* imaxz) {
393 const float kdist = static_cast<float>(r);
394 *iminx = lower_shift(c[0], kdist, ox, sp, nx);
395 *iminy = lower_shift(c[1], kdist, oy, sp, ny);
396 *iminz = lower_shift(c[2], kdist, oz, sp, nz);
397 *imaxx = upper_shift(c[0], kdist, ox, sp, nx);
398 *imaxy = upper_shift(c[1], kdist, oy, sp, ny);
399 *imaxz = upper_shift(c[2], kdist, oz, sp, nz);
